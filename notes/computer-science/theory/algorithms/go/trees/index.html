func preorderTraversal(root *TreeNode) []int {
	var ret []int
	preorderHelper(root, &ret)
	return ret
}

func preorderHelper(cur *TreeNode, ret *[]int) {
	if cur == nil {
		return
	}
	*ret = append(*ret, cur.Val)
	preorderHelper(cur.Left, ret)
	preorderHelper(cur.Right, ret)
}

迭代

使用栈模拟递归

func preorderTraversal(root *TreeNode) []int {
	var ret []int
	var stack []*TreeNode
	var top int = 0
	if root == nil {
		return ret
	}

	stack = append(stack, root)
	top++
	for top > 0 {
		cur := stack[top-1]
		stack = stack[:top-1]
		top--
		ret = append(ret, cur.Val)
		if cur.Right != nil {
			stack = append(stack, cur.Right)
			top++
		}
		if cur.Left != nil {
			stack = append(stack, cur.Left)
			top++
		}
	}
	return ret
}

二叉树的层平均值

Given the root of a binary tree, return the average value of the nodes on each level in the form of an array. Answers within 10-5 of the actual answer will be accepted.

func averageOfLevels(root *TreeNode) []float64 {
	var ret []float64
	var queue []*TreeNode

	queue = append(queue, root)
	for size := len(queue); size > 0; size = len(queue) {
		var sum int
		for range size {
			cur := queue[0]
			queue = queue[1:]
			sum += cur.Val
			if cur.Left != nil {
				queue = append(queue, cur.Left)
			}
			if cur.Right != nil {
				queue = append(queue, cur.Right)
			}
		}
		ret = append(ret, float64(sum)/float64(size))
	}
	return ret
}

func largestValues(root *TreeNode) []int {

		return ret

			}

				queue = append(queue, cur.Right)
			}
		}

	}

}

填充每个节点的下一个右侧节点指针

Populate each next pointer to point to its next right node. If there is no next right node, the next pointer should be set to NULL.

Initially, all next pointers are set to NULL.

func connect(root *Node) *Node {
	if root == nil {
		return root
	}
	var queue []*Node
	queue = append(queue, root)

	for size := len(queue); size > 0; size = len(queue) {
		var prev *Node = nil
		for range size {
			cur := queue[0]
			queue = queue[1:]

			if prev != nil {
				prev.Next = cur
			}
			prev = cur

			if cur.Left != nil {
				queue = append(queue, cur.Left)
			}
			if cur.Right != nil {
				queue = append(queue, cur.Right)
			}
		}
	}

	return root
}

翻转二叉树

Given the root of a binary tree, invert the tree, and return its root.

  • 前序：先交换 Left 和 Right，再递归交换左子树和右子树
  • 后序：先递归交换左子树和右子树，最后交换 Left 和 Right
func invertTree(root *TreeNode) *TreeNode {
	if root == nil {
		return root
	}

	root.Left, root.Right = invertTree(root.Right), invertTree(root.Left)
	return root
}

Given the root of a binary tree, check whether it is a mirror of itself (i.e., symmetric around its center).

前序遍历实现

func isSymmetric(root *TreeNode) bool {
	if root == nil {
		return true
	}
	return isSymmetricHelper(root.Left, root.Right)
}

func isSymmetricHelper(left, right *TreeNode) bool {
	if left == nil && right == nil {
		return true
	} else if left != nil && right == nil || left == nil && right != nil {
		return false
	}

	return left.Val == right.Val && isSymmetricHelper(left.Left, right.Right) && isSymmetricHelper(left.Right, right.Left)
}

二叉树的最大深度

Given the root of a binary tree, return its maximum depth.

A binary tree’s maximum depth is the number of nodes along the longest path from the root node down to the farthest leaf node.

func maxDepth(root *TreeNode) int {
	if root == nil {
		return 0
	}
	return 1 + max(maxDepth(root.Left), maxDepth(root.Right))
}

N 叉树的最大深度

Given a n-ary tree, find its maximum depth.

The maximum depth is the number of nodes along the longest path from the root node down to the farthest leaf node.

Nary-Tree input serialization is represented in their level order traversal, each group of children is separated by the null value (See examples).

func maxDepth(root *Node) int {
	if root == nil {
		return 0
	}
	theMax := 0
	for _, node := range root.Children {
		theMax = max(theMax, maxDepth(node))
	}
	return theMax + 1
}

二叉树的最小深度

Given a binary tree, find its minimum depth.

The minimum depth is the number of nodes along the shortest path from the root node down to the nearest leaf node.

Note: A leaf is a node with no children

比起最大深度，需要额外判断当前是否为叶子结点

func minDepth(root *TreeNode) int {
	if root == nil {
		return 0
	} else if root.Left == nil && root.Right == nil {
		return 1
	}
	var curMin int = math.MaxInt
	if root.Left != nil {
		curMin = min(curMin, minDepth(root.Left))
	}
	if root.Right != nil {
		curMin = min(curMin, minDepth(root.Right))
	}

	return 1 + curMin
}

平衡二叉树

Given a binary tree, determine if it is height-balanced.

func isBalanced(root *TreeNode) bool {
	_, balanced := isBalancedHelper(root)
	return balanced
}

func isBalancedHelper(root *TreeNode) (int, bool) {
	if root == nil {
		return 0, true
	}
	leftHeight, leftBalanced := isBalancedHelper(root.Left)
	rightHeight, rightBalanced := isBalancedHelper(root.Right)

	if !leftBalanced || !rightBalanced || leftHeight-rightHeight > 1 || rightHeight-leftHeight > 1 {
		return 0, false
	}

	return max(leftHeight, rightHeight) + 1, true
}

完全二叉树

Given the root of a complete binary tree, return the number of the nodes in the tree.

According to Wikipedia, every level, except possibly the last, is completely filled in a complete binary tree, and all nodes in the last level are as far left as possible. It can have between 1 and 2h nodes inclusive at the last level h.

Design an algorithm that runs in less than O(n) time complexity.

We can just iterate the whole tree directly while it will cost O(n)O(n)O(n), when the question asks us to run in less than O(n)O(n)O(n). So we should use the trait of a complete binary tree.

Every level, except possibly the last, is completely filled in a complete binary tree, and all nodes in the last level are as far left as possible. It can have between 111 and 2n2^n2n nodes inclusive at the last level nnn.

So there are only 2 stituations for complete binary tree:

  1. The tree is full, then the 2h−12^h-12h−1 is the answer.
  2. The last level of the tree is not full, while we iterate the left subtree and the right subtree of the root. Then there is once a time that the subtree will be full which we can use the formula above to calculate.

So how to judge whether the subtree is full?

  • Iterate from cur node through the left and the right.
  • Compare the lengths of each of the two directions.
    • If the they are same, the subtree which root is cur is full.
    • Else, iterate both of the subtree
func countNodes(root *TreeNode) int {
	var ret int = 0
	if root == nil {
		return ret
	}

	var maxLeft, maxRight int = 0, 0
	for l := root; l != nil; l = l.Left {
		maxLeft++
	}
	for r := root; r != nil; r = r.Right {
		maxRight++
	}

	if maxLeft == maxRight {
		return (1 << maxLeft) - 1
	}
	return countNodes(root.Left) + countNodes(root.Right) + 1
}

二叉树的所有路径

Given the root of a binary tree, return *all root-to-leaf paths in any order*.

A leaf is a node with no children.

const Arrow = "->"

func binaryTreePaths(root *TreeNode) []string {
	var ret []string
	if root == nil {
		return ret
	}
	var path []string
	dfs(root, &path, &ret)
	return ret
}

func dfs(cur *TreeNode, path *[]string, ret *[]string) {
	*path = append(*path, strconv.Itoa(cur.Val))
	if cur.Left == nil && cur.Right == nil {
		*ret = append(*ret, strings.Join(*path, Arrow))
		*path = (*path)[:len(*path)-1]
		return
	}
	if cur.Left != nil {
		dfs(cur.Left, path, ret)
	}
	if cur.Right != nil {
		dfs(cur.Right, path, ret)
	}
	*path = (*path)[:len(*path)-1]
}

左叶子之和

Given the root of a binary tree, return the sum of all left leaves.

A leaf is a node with no children. A left leaf is a leaf that is the left child of another node.

The key is to judge whether the current node is the left leaf. So we use the flag isLeft to mark.

  • When it is true, means we are iterating through the left subtree.
  • When it is false, means we are go through the right subtree.

When we reach the leaf node, then judge whether it is the left leaf with isLeft.

func sumOfLeftLeaves(root *TreeNode) int {
	var ret int = 0
    if root == nil || (root.Left == nil && root.Right == nil) {
		return ret
	}
	return leftLeavesDfs(root, false)
}

func leftLeavesDfs(cur *TreeNode, isLeft bool) int {
	if isLeft && cur.Left == nil && cur.Right == nil {
		return cur.Val
	}
	var sum int = 0
	if cur.Left != nil {
		sum += leftLeavesDfs(cur.Left, true)
	}
	if cur.Right != nil {
		sum += leftLeavesDfs(cur.Right, false)
	}
	return sum
}

找树左下角的值

Given the root of a binary tree, return the leftmost value in the last row of the tree.

  • Find the deepest row
  • Use preorder iterating, the first node we meet in one row is the leftmost node.
var maxDepth, leftValue int

func findBottomLeftValue(root *TreeNode) int {
	maxDepth, leftValue = 1, root.Val
	bottomLeftDfs(root, 1)
	return leftValue
}

func bottomLeftDfs(cur *TreeNode, depth int) {
	if cur == nil {
		return
	}
	if cur.Left == nil && cur.Right == nil {
		if depth > maxDepth {
			maxDepth = depth
			leftValue = cur.Val
		}
		return
	}

	bottomLeftDfs(cur.Left, depth+1)
	bottomLeftDfs(cur.Right, depth+1)
}

路径总和

Given the root of a binary tree and an integer targetSum, return true if the tree has a root-to-leaf path such that adding up all the values along the path equals targetSum.

func hasPathSum(root *TreeNode, targetSum int) bool {
	if root == nil {
		return false
	}
	sum := targetSum - root.Val
	if root.Left == nil && root.Right == nil && sum == 0 {
		return true
	}

	var left, right bool = false, false
	if root.Left != nil {
		left = hasPathSum(root.Left, sum)
	}
	if root.Right != nil {
		right = hasPathSum(root.Right, sum)
	}

	return left || right
}

树的构造

从中序与后序遍历序列构造二叉树

Given two integer arrays inorder and postorder where inorder is the inorder traversal of a binary tree and postorder is the postorder traversal of the same tree, construct and return the binary tree.

func buildTree(inorder []int, postorder []int) *TreeNode {
	return buildTreeHelper(inorder, postorder, 0, len(inorder)-1, 0, len(postorder)-1)
}

func buildTreeHelper(inorder, postorder []int, il, ir, pl, pr int) *TreeNode {
	if il > ir {
		return nil
	}
	var curVal = postorder[pr]
	var cur *TreeNode = &TreeNode{
		Val: curVal,
	}

	var curAppend int
	for curAppend = 0; il+curAppend <= ir; curAppend++ {
		if inorder[il+curAppend] == curVal {
			break
		}
	}

	cur.Left = buildTreeHelper(inorder, postorder, il, il+curAppend-1, pl, pl+curAppend-1)
	cur.Right = buildTreeHelper(inorder, postorder, il+curAppend+1, ir, pl+curAppend, pr-1)
	return cur
}

最大二叉树

You are given an integer array nums with no duplicates. A maximum binary tree can be built recursively from nums using the following algorithm:

  1. Create a root node whose value is the maximum value in nums.
  2. Recursively build the left subtree on the subarray prefix to the left of the maximum value.
  3. Recursively build the right subtree on the subarray suffix to the right of the maximum value.

Return the maximum binary tree built from nums.

func constructMaximumBinaryTree(nums []int) *TreeNode {
	return constructMaximumBinaryTreeHelper(nums, 0, len(nums)-1)
}

func constructMaximumBinaryTreeHelper(nums []int, ml, mr int) *TreeNode {
	if ml > mr {
		return nil
	} else if ml == mr {
		return &TreeNode{
			Val: nums[ml],
		}
	}
	var maxIdx int = ml
	var maxVal int = nums[ml]
	for i := ml + 1; i <= mr; i++ {
		if maxVal < nums[i] {
			maxIdx = i
			maxVal = nums[i]
		}
	}

	return &TreeNode{
		Val:   maxVal,
		Left:  constructMaximumBinaryTreeHelper(nums, ml, maxIdx-1),
		Right: constructMaximumBinaryTreeHelper(nums, maxIdx+1, mr),
	}
}

合并二叉树

You are given two binary trees root1 and root2.

Imagine that when you put one of them to cover the other, some nodes of the two trees are overlapped while the others are not. You need to merge the two trees into a new binary tree. The merge rule is that if two nodes overlap, then sum node values up as the new value of the merged node. Otherwise, the NOT null node will be used as the node of the new tree.

Return the merged tree.

Note: The merging process must start from the root nodes of both trees.

func mergeTrees(root1 *TreeNode, root2 *TreeNode) *TreeNode {
	if root1 == nil && root2 == nil {
		return nil
	} else if root1 == nil {
		return root2
	} else if root2 == nil {
		return root1
	}
	return &TreeNode{
		Val:   root1.Val + root2.Val,
		Left:  mergeTrees(root1.Left, root2.Left),
		Right: mergeTrees(root1.Right, root2.Right),
	}
}

二叉搜索树中的搜索

You are given the root of a binary search tree (BST) and an integer val.

Find the node in the BST that the node’s value equals val and return the subtree rooted with that node. If such a node does not exist, return null.

func searchBST(root *TreeNode, val int) *TreeNode {
	if root == nil {
		return nil
	}
	if root.Val == val {
		return root
	} else if root.Val < val {
		return searchBST(root.Right, val)
	} else {
		return searchBST(root.Left, val)
	}
}

验证二叉搜索树

Given the root of a binary tree, determine if it is a valid binary search tree (BST).

A valid BST is defined as follows:

  • The left subtree of a node contains only nodes with keys strictly less than the node’s key.
  • The right subtree of a node contains only nodes with keys strictly greater than the node’s key.
  • Both the left and right subtrees must also be binary search trees.

preorder

func isValidBST(root *TreeNode) bool {
	return isValidBSTHelper(root, math.MinInt, math.MaxInt)
}

func isValidBSTHelper(root *TreeNode, minLimit, maxLimit int) bool {
	if root == nil {
		return true
	}
	if root.Val <= minLimit || root.Val >= maxLimit {
		return false
	}
	return isValidBSTHelper(root.Left, minLimit, min(root.Val, maxLimit)) && isValidBSTHelper(root.Right, max(root.Val, minLimit), maxLimit)
}
