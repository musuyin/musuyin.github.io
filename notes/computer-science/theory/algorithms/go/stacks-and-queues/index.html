阅读线索

栈和队列的处理顺序不同，适合不同的待处理信息。括号和表达式用栈保留尚未匹配的结构，单调队列和堆则保留可能成为最优答案的候选。

	s.size++

	ret := s.arr[s.size-1]

栈

用栈实现队列

Implement a first in first out (FIFO) queue using only two stacks. The implemented queue should support all the functions of a normal queue (push, peek, pop, and empty).

Implement the MyQueue class:

  • void push(int x) Pushes element x to the back of the queue.
  • int pop() Removes the element from the front of the queue and returns it.
  • int peek() Returns the element at the front of the queue.
  • boolean empty() Returns true if the queue is empty, false otherwise.

核心是一个栈可以将一个输入顺序反转，两个栈就能保持输入顺序不变。所以使用两个栈

type stack struct {
	arr  []int
	size int
}

func NewStack() stack {
	return stack{
		arr:  []int{},
		size: 0,
	}
}

func (s *stack) push(x int) {
	s.arr = append(s.arr, x)
	s.size++
}

func (s *stack) pop() int {
	ret := s.arr[s.size-1]
	s.arr = s.arr[:s.size-1]
	s.size--
	return ret
}

func (s *stack) top() int {
	return s.arr[s.size-1]
}

func (s *stack) isEmpty() bool {
	return s.size == 0
}

type MyQueue struct {
	in stack
	out stack
}

func Constructor() MyQueue {
	return MyQueue{
		in: NewStack(),
		out: NewStack(),
	}
}

func (this *MyQueue) Push(x int) {
	this.in.push(x)
}

func (this *MyQueue) convert() {
	for !this.in.isEmpty() {
		this.out.push(this.in.pop())
	}
}

func (this *MyQueue) Pop() int {
	if this.out.isEmpty() {
		this.convert()
	}
	return this.out.pop()
}

func (this *MyQueue) Peek() int {
	if this.out.isEmpty() {
		this.convert()
	}
	return this.out.top()
}

func (this *MyQueue) Empty() bool {
	return this.in.isEmpty() && this.out.isEmpty()
}

有效的括号

Given a string s containing just the characters '(', ')', '{', '}', '[' and ']', determine if the input string is valid.

An input string is valid if:

  1. Open brackets must be closed by the same type of brackets.
  2. Open brackets must be closed in the correct order.
  3. Every close bracket has a corresponding open bracket of the same type.
type CharStack struct {
	arr  []byte
	size int
}

func NewCharStack() CharStack {
	return CharStack{
		arr:  []byte{},
		size: 0,
	}
}

func (s *CharStack) push(x byte) {
	s.arr = append(s.arr, x)
	s.size++
}

func (s *CharStack) pop() byte {
	ret := s.arr[s.size-1]
	s.arr = s.arr[:s.size-1]
	s.size--
	return ret
}

func (s *CharStack) top() byte {
	return s.arr[s.size-1]
}

func (s *CharStack) isEmpty() bool {
	return s.size == 0
}

func isValid(s string) bool {
	st := NewCharStack()
	for i := 0; i < len(s); i++ {
		c := s[i]
		switch c {
		case '(':
			st.push(')')
		case '[':
			st.push(']')
		case '{':
			st.push('}')
		default:
			if st.isEmpty() || c != st.pop() {
				return false
			}
		}
	}
	return st.isEmpty()
}

删除字符串中所有相邻重复项

You are given a string s consisting of lowercase English letters. A duplicate removal consists of choosing two adjacent and equal letters and removing them.

We repeatedly make duplicate removals on s until we no longer can.

Return the final string after all such duplicate removals have been made. It can be proven that the answer is unique.

  • 遇到数字存入栈
  • 遇到符号取出两个数字进行运算，结果存回栈

注意负数和 - 的区分

type CharStack struct {
	arr  []byte
	size int
}

func NewCharStack() CharStack {
	return CharStack{
		arr:  []byte{},
		size: 0,
	}
}

func (s *CharStack) push(x byte) {
	s.arr = append(s.arr, x)
	s.size++
}

func (s *CharStack) pop() byte {
	ret := s.arr[s.size-1]
	s.arr = s.arr[:s.size-1]
	s.size--
	return ret
}

func (s *CharStack) top() byte {
	return s.arr[s.size-1]
}

func (s *CharStack) isEmpty() bool {
	return s.size == 0
}

func removeDuplicates(s string) string {
	st := NewCharStack()
	for i := 0; i < len(s); i++ {
		c := s[i]
		if !st.isEmpty() && c == st.top() {
			st.pop()
			continue
		}
		st.push(c)
	}

	retCs := make([]byte, st.size)
	for i := st.size - 1; i >= 0; i-- {
		retCs[i] = st.pop()
	}
	return string(retCs)
}

逆波兰表达式求值

You are given an array of strings tokens that represents an arithmetic expression in a Reverse Polish Notation.

Evaluate the expression. Return an integer that represents the value of the expression.

import "strconv"

type IntStack struct {
	arr  []int
	size int
}

func NewIntStack() IntStack {
	return IntStack{
		arr:  []int{},
		size: 0,
	}
}

func (this *IntStack) push(x int) {
	this.arr = append(this.arr, x)
	this.size++
}

func (this *IntStack) pop() int {
	ret := this.arr[this.size-1]
	this.arr = this.arr[:this.size-1]
	this.size--
	return ret
}

func isDigit(s string) bool {
	if len(s) > 1 || ('0' <= s[0] && s[0] <= '9') {
		return true
	}
	return false
}

func evalRPN(tokens []string) int {
	st := NewIntStack()
	for _, token := range tokens {
		if isDigit(token) {
			num, _ := strconv.Atoi(token)
			st.push(num)
			continue
		}

		firstChar := token[0]
		switch firstChar {
		case '+':
			a := st.pop()
			b := st.pop()
			st.push(b + a)
		case '-':
			a := st.pop()
			b := st.pop()
			st.push(b - a)
		case '*':
			a := st.pop()
			b := st.pop()
			st.push(b * a)
		case '/':
			a := st.pop()
			b := st.pop()
			st.push(b / a)
		}
	}
	return st.pop()
}

队列

用队列实现栈

队列会维持输入顺序，所以不能通过数据结构特性来实现，只能先移出前 size()-1 个元素，输出最后一个元素。用 1 或 2 个队列都可实现

type queue struct {
	arr  []int
	size int
}

func NewQueue() queue {
	return queue{
		arr:  []int{},
		size: 0,
	}
}

func (q *queue) push(x int) {
	q.arr = append(q.arr, x)
	q.size++
}

func (q *queue) pop() int {
	ret := q.arr[0]
	q.arr = q.arr[1:]
	q.size--
	return ret
}

func (q *queue) top() int {
	return q.arr[0]
}

func (q *queue) isEmpty() bool {
	return q.size == 0
}

type MyStack struct {
	firstQueue  queue
	secondQueue queue
	flag        bool
}

func Constructor() MyStack {
	return MyStack{
		firstQueue:  queue{},
		secondQueue: queue{},
		flag:        false,
	}
}

func (this *MyStack) Push(x int) {
	if this.flag {
		this.firstQueue.push(x)
	} else {
		this.secondQueue.push(x)
	}
}

func (this *MyStack) Pop() int {
	var ret int
	if this.flag {
		sz := this.firstQueue.size - 1
		for i := 0; i < sz; i++ {
			this.secondQueue.push(this.firstQueue.pop())
		}
		ret = this.firstQueue.pop()
	} else {
		sz := this.secondQueue.size - 1
		for i := 0; i < sz; i++ {
			this.firstQueue.push(this.secondQueue.pop())
		}
		ret = this.secondQueue.pop()
	}
	this.flag = !this.flag
	return ret
}

func (this *MyStack) Top() int {
	var ret int
	if this.flag {
		sz := this.firstQueue.size - 1
		for i := 0; i < sz; i++ {
			this.secondQueue.push(this.firstQueue.pop())
		}
		ret = this.firstQueue.pop()
		this.secondQueue.push(ret)
	} else {
		sz := this.secondQueue.size - 1
		for i := 0; i < sz; i++ {
			this.firstQueue.push(this.secondQueue.pop())
		}
		ret = this.secondQueue.pop()
		this.firstQueue.push(ret)
	}
	this.flag = !this.flag
	return ret
}

func (this *MyStack) Empty() bool {
	return this.firstQueue.isEmpty() && this.secondQueue.isEmpty()
}

单调队列

滑动窗口最大值

You are given an array of integers nums, there is a sliding window of size k which is moving from the very left of the array to the very right. You can only see the k numbers in the window. Each time the sliding window moves right by one position.

Return the max sliding window.

  1. 维护队列单调递减，每轮取出队首元素
  2. 还需要判断队首元素是否超出窗口范围，如果超出还要出队，但是每轮最多只会出队一个元素，所以进行一次更新即可
  3. 更新后当前队首元素即为窗口中最大值
type node struct {
	idx int
	val int
}

func (this *node) less(x node) bool {
	return this.val < x.val
}

type MonoDescQueue struct {
	arr  []*node
	size int
}

func NewMonoDescQueue() MonoDescQueue {
	return MonoDescQueue{
		arr:  []*node{},
		size: 0,
	}
}

func (this *MonoDescQueue) Push(x *node) {
	for this.size > 0 && this.arr[this.size-1].less(*x) {
		this.arr = this.arr[:this.size-1]
		this.size--
	}
	this.arr = append(this.arr, x)
	this.size++
}

func (this *MonoDescQueue) Pop() *node {
	if this.size == 0 {
		return nil
	}
	ret := this.arr[0]
	this.arr = this.arr[1:]
	this.size--
	return ret
}

func (this *MonoDescQueue) Top() *node {
	if this.size == 0 {
		return nil
	}
	return this.arr[0]
}

func maxSlidingWindow(nums []int, k int) []int {
	size := len(nums)
	queue := NewMonoDescQueue()
	ret := make([]int, size-k+1)
	for i, num := range nums {
		// insert into queue
		queue.Push(&node{i, num})
		// do not judge the first k node
		if i < k-1 {
			continue
		}

		// check whether the first node in queue is expired
		// only one or zero node will be removed every round
		top := queue.Top()
		if i-k >= top.idx {
			queue.Pop()
		}
		// update the current biggest node
		top = queue.Top()
		ret[i-k+1] = top.val
	}
	return ret
}

优先队列

前 K 个高频元素

Given an integer array nums and an integer k, return the k most frequent elements. You may return the answer in any order.

难点主要是 Go 中没有原生优先队列实现，需要手搓。关键在于 swim() 和 sink()

  • swim()：当前结点向上浮，判断该结点的父结点（cur/2）与自身大小
  • sink()：当前结点向下沉，该结点与其一个或两个子结点进行大小判断
    • cur*2
    • cur*2+1

两个操作都是要进行直到不可进行为止

优先队列操作：

  • Push()：向数组末尾插入元素，使用 swim() 进行上浮
  • Pop()：将数组首元素与末尾元素交换，并删除更换后的末尾元素（即 size--），交换后对于首元素，使用 sink() 进行下沉
type node struct {
	val   int
	times int
}

func (this *node) less(x *node) bool {
	return this.times < x.times
}

type PriorityQueue struct {
	arr      []*node
	size     int
	capacity int
}

func NewPriorityQueue(capacity int) PriorityQueue {
	return PriorityQueue{
		arr:      make([]*node, capacity+1),
		size:     0,
		capacity: capacity,
	}
}

func (this *PriorityQueue) Push(x *node) {
	this.size++
	this.arr[this.size] = x
	this.swim()
}

func (this *PriorityQueue) Pop() *node {
	if this.size == 0 {
		return nil
	}
	ret := this.arr[1]
	this.swap(1, this.size)
	this.size--
	this.sink(1)
	return ret
}

func (this *PriorityQueue) Top() *node {
	if this.size == 0 {
		return nil
	}
	return this.arr[1]
}

func (this *PriorityQueue) swim() {
	cur := this.size
	for cur/2 > 0 && this.arr[cur/2].less(this.arr[cur]) {
		this.swap(cur, cur/2)
		cur = cur / 2
	}
}

func (this *PriorityQueue) sink(x int) {
	cur := x
	nxt := cur * 2
	if nxt <= this.size && this.arr[cur].less(this.arr[nxt]) {
		cur = nxt
	}
	if nxt+1 <= this.size && this.arr[cur].less(this.arr[nxt+1]) {
		cur = nxt + 1
	}
	if cur != x {
		this.swap(x, cur)
		this.sink(cur)
	}
}

func (this *PriorityQueue) swap(a, b int) {
	this.arr[a], this.arr[b] = this.arr[b], this.arr[a]
}

func topKFrequent(nums []int, k int) []int {
	times := make(map[int]int)
	for _, num := range nums {
		times[num]++
	}
	pq := NewPriorityQueue(len(times))
	for val, times := range times {
		node := &node {
			val:   val,
			times: times,
		}
		pq.Push(node)
	}

	ret := make([]int, k)
	for i := 0; i < k; i++ {
		ret[i] = pq.Pop().val
	}
	return ret
}
