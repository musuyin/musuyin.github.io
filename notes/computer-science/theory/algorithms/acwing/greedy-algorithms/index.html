贪心算法

2023-12-14

#算法与数据结构#CPP#Acwing

阅读线索

贪心解法需要把局部选择与全局目标连接起来。区间问题关注排序与边界，Huffman 关注重复付出的代价，排序不等式则解释相邻交换为什么有利。

贪心算法的难点在于证明贪心策略的正确性。只有当问题具有单峰性（即局部最优解能推导出全局最优解）时，才适合使用贪心方法。

区间问题

区间问题的常见思路是尝试按左端点或右端点排序，再结合具体场景加以证明。

区间选点

  1. 将每个区间按右端点从小到大排序

  2. 从前往后依次枚举每个区间

    • 如果当前区间已经包含点，则跳过
    • 否则，选择当前区间的右端点

证明：参考思路：a≤ba \leq ba≤b 且 a≥ba \geq ba≥b，则 a=ba = ba=b

#include <iostream>
#include <algorithm>
using namespace std;

const int N = 100010;

int n;
struct Range
{
    int l, r;
} range[N];

int cmp(Range a, Range b)
{
    return a.r < b.r;  //注意cpp中cmp与c中cmp相反
}

int main(void)
{
    scanf("%d", &n);
    for (int i = 0; i < n; i++)
        scanf("%d %d", &range[i].l, &range[i].r);

    sort(range, range + n, cmp);

    int res = 0, ed = -2e9;  //ed表示当前最右的端点位置
    for (int i = 0; i < n; i++)
        if (range[i].l > ed)  //如果新区间的最左段都在当前最右端点的右边，说明要加入新的点了
        {
            res++;
            ed = range[i].r;
        }

    printf("%d\n", res);

    return 0;
}

最大不相交区间数量

思路同上一题，以右端点排序。

#include <iostream>
#include <algorithm>
using namespace std;

const int N = 100010;
int n;
struct Range
{
    int l, r;
} range[N];

int cmp(Range a, Range b)
{
    return a.r < b.r;
}

int main(void)
{
    scanf("%d", &n);
    for (int i = 0; i < n; i++)
        scanf("%d %d", &range[i].l, &range[i].r);

    sort(range, range + n, cmp);

    int res = 0, ed = -2e9;

        {

区间分组

  1. 以左端点排序

  2. 从前往后处理每个区间

    • 判断当前区间能否放到某个现有的组中：该组的最右端点是否小于当前区间的左端点，即 Max_r<L[i]\text{Max_r} < L[i]Max_r<L[i]
    • 如果不存在这样的组，则开新的组并放入该区间
    • 若存在这样的组，则放入该区间并更新当前组的 Max_r\text{Max_r}Max_r

证明：目标为证明 ans=cnt\text{ans} = \text{cnt}ans=cnt

  1. ans≤cnt\text{ans} \leq \text{cnt}ans≤cnt：按照上述算法得到一种合法方案 cnt\text{cnt}cnt，而 ans\text{ans}ans 是所有合法方案中的最小值，故 ans≤cnt\text{ans} \leq \text{cnt}ans≤cnt
  2. ans≥cnt\text{ans} \geq \text{cnt}ans≥cnt：假设现在有 cnt−1\text{cnt} - 1cnt−1 个组，当新加入一个区间时，现存组与新区间均发生重合，说明新区间的左端点 L[i]L[i]L[i] 被前 cnt−1\text{cnt} - 1cnt−1 个组的右端点覆盖，必须创建新组；此时共 cnt\text{cnt}cnt 个组之间必然存在两两相交，因此至少需要 cnt\text{cnt}cnt 个组，即 ans≥cnt\text{ans} \geq \text{cnt}ans≥cnt
#include <iostream>
#include <algorithm>
#include <queue>
using namespace std;

const int N = 100010;

int n;
struct Range
{
    int l, r;
    bool operator< (const Range &W)const
    {
        return l < W.l;
    }
} range[N];

int main(void)
{
    scanf("%d", &n);
    for (int i = 0; i < n; i++)
    {
        int l, r;
        scanf("%d %d", &l, &r);
        range[i] = {l, r};
    }

    sort(range, range + n);

    priority_queue<int, vector<int>, greater<int> > heap;  //使用小根堆维护当前所有组的右端点
    for (int i = 0; i < n; i++)
    {
        auto r = range[i];
        if (heap.empty() || heap.top() >= r.l)  //堆为空；所有组中最小的右端点都大于当前区间，则要开新组
            heap.push(r.r);
        else
        {
            heap.pop();  //更新当前的右端点
            heap.push(r.r);  //堆会自行维护
        }
    }

    printf("%d\n", heap.size());

    return 0;
}

区间覆盖

  1. 将所有区间按左端点从小到大排序

  2. 从前往后依次枚举每个区间

    • 选择左端点可以覆盖线段起点 start\text{start}start 的、右端点最大的区间（注意是右端点最大，而非区间最长）
    • 更新 start\text{start}start 为该区间的右端点，重复执行直至 start≥end\text{start} \geq \text{end}start≥end

证明：

  1. ans≥cnt\text{ans} \geq \text{cnt}ans≥cnt：反证法，找与预设矛盾的点；对于每一种最优方案 ans\text{ans}ans，都能通过合法变换转化为 cnt\text{cnt}cnt，故两者等价
  2. ans≤cnt\text{ans} \leq \text{cnt}ans≤cnt
#include <iostream>
#include <algorithm>
using namespace std;

const int N = 100010;

int n;
struct Range
{
    int l, r;
    bool operator< (const Range &W)const
    {
        return l < W.l;
    }
} range[N];

int main(void)
{
    int st, ed;
    scanf("%d %d", &st, &ed);
    scanf("%d", &n);
    for (int i = 0; i < n; i++)
    {
        int l, r;
        scanf("%d %d", &l, &r);
        range[i] = {l, r};
    }

    sort(range, range + n);

    int res = 0;
    bool success = false;
    for (int i = 0; i < n; i++)
    {   //双指针
        int j = i, r = -2e9;
        while (j < n && range[j].l <= st)  //找到左端点在st左侧的右端点最大的区间
        {
            r = max(r, range[j].r);
            j++;
        }

        if (r < st)  //如果发现现存区间的最右端已经小于st了，说明无法覆盖区间
        {
            res = -1;
            break;
        }

        res++;  //所需区间数++
        if (r >= ed)  //成功覆盖整个线段
        {
            success = true;
            break;
        }

        st = r;  //更新
        i = j - 1;
    }

    if (!success) res = -1;
    printf("%d\n", res);

    return 0;
}

Huffman 树

Huffman 树（哈夫曼树）是一种经典的贪心构造，其核心思想是每次合并权值最小的两个节点。

合并果子

注：区间 DP 只能合并相邻两堆（如合并石子），本题无视距离限制。

哈夫曼树策略：每次合并两个最小的堆，将合并所得结果作为新节点放入。

证明：

  1. 在所有节点中，权值最小的两个点深度一定最深，且可以互为兄弟

    反证法：

    一定最深：假设 aaa 和 fff 最小，则交换 bbb 和 fff 后总值一定减小，说明最优解中最小的两个点深度一定最深。

    互为兄弟：若有多个权值相同的最小节点，可相互交换而不影响总值。

  2. F(n−1)F(n-1)F(n−1) 的最优解也是 F(n)F(n)F(n) 的最优解

    设 f(n)f(n)f(n) 是 F(n)F(n)F(n) 的一种解法，则 f(n)=f(n−1)+a+bf(n) = f(n-1) + a + bf(n)=f(n−1)+a+b。所有求 f(n)f(n)f(n) 的式子都会加上 a+ba + ba+b，因此 min⁡(f(n))\min(f(n))min(f(n)) 等价于 min⁡(f(n−1))\min(f(n-1))min(f(n−1))。

#include <iostream>
#include <queue>
#include <algorithm>
using namespace std;

int main(void)
{
    int n;
    scanf("%d", &n);

    priority_queue<int, vector<int>, greater<int> > heap;
    while (n--)
    {
        int x;
        scanf("%d", &x);
        heap.push(x);
    }

    int res = 0;
    while (heap.size() > 1)
    {
        int a = heap.top(); heap.pop();  //取出最小的两个数
        int b = heap.top(); heap.pop();
        res += a + b;
        heap.push(a + b);
    }

    printf("%d\n", res);
    return 0;
}

排序不等式

排队打水

  1. 按所需时间从小到大排序
  2. 计算总等待时间

证明：反证法。

假设序列不按升序排列，则必然存在至少一对逆序对 ti>ti+1t_i > t_{i+1}ti​>ti+1​。

交换前：ti×(n−i)+ti+1×(n−i−1)t_i \times (n - i) + t_{i+1} \times (n - i - 1)ti​×(n−i)+ti+1​×(n−i−1)

交换后：ti×(n−i−1)+ti+1×(n−i)t_i \times (n - i - 1) + t_{i+1} \times (n - i)ti​×(n−i−1)+ti+1​×(n−i)

交换后总时间更小，说明存在更优解，因此序列中不能存在逆序对。

总时间公式为：

T=t1×(n−1)+t2×(n−2)+⋯+tn×0T = t_1 \times (n-1) + t_2 \times (n-2) + \cdots + t_n \times 0T=t1​×(n−1)+t2​×(n−2)+⋯+tn​×0

#include <iostream>
#include <algorithm>
using namespace std;

const int N = 100010;
int t[N];

typedef long long LL;

int main(void)
{
    int n;
    cin >> n;
    for (int i = 0; i < n; i++)
        scanf("%d", &t[i]);

    sort(t, t + n);

    LL res = 0;
    for (int i = 0; i < n; i++)
        res += t[i] * (n - i - 1);

    printf("%lld", res);
}

货仓选址

#include <iostream>
#include <algorithm>
using namespace std;

const int N = 100010;

int n;
int q[N];

int main(void)
{
    scanf("%d", &n);
    for (int i = 0; i < n; i++)
        scanf("%d", &q[i]);

    sort(q, q + n);

    int res = 0;

        res += abs(q[i] - q[n / 2]);  //相当于已经选好点在中间了，然后直接算距离

耍杂技的牛

按照 wi+siw_i + s_iwi​+si​ 从小到大顺序排序，可得最优解。

证明：反证法。

  1. 贪心得到的答案≥最优解\text{贪心得到的答案} \geq \text{最优解}贪心得到的答案≥最优解（显然）
  2. 贪心得到的答案≤最优解\text{贪心得到的答案} \leq \text{最优解}贪心得到的答案≤最优解

假设最优解不按上述顺序排列，则必然存在相邻的一对满足 wi+si>wi+1+si+1w_i + s_i > w_{i+1} + s_{i+1}wi​+si​>wi+1​+si+1​。

考虑位置为 iii 和 i+1i+1i+1 的两头牛：

交换前：i 号：w1+w2+⋯+wi−1−siw_1 + w_2 + \cdots + w_{i-1} - s_iw1​+w2​+⋯+wi−1​−si​；i+1 号：w1+w2+⋯+wi−si+1w_1 + w_2 + \cdots + w_i - s_{i+1}w1​+w2​+⋯+wi​−si+1​

交换后：i 号：w1+w2+⋯+wi−1−si+1w_1 + w_2 + \cdots + w_{i-1} - s_{i+1}w1​+w2​+⋯+wi−1​−si+1​；i+1 号：w1+w2+⋯+wi−1+wi+1−siw_1 + w_2 + \cdots + w_{i-1} + w_{i+1} - s_iw1​+w2​+⋯+wi−1​+wi+1​−si​

去掉公共项：

i 号危险系数 i+1 号危险系数
交换前 −si+1-s_{i+1}−si+1​ wi−siw_i - s_iwi​−si​
交换后 −si-s_i−si​ wi+1−si+1w_{i+1} - s_{i+1}wi+1​−si+1​

注意：题目中危险系数为各奶牛所受压力减去自身强壮值，表中已省略公共前缀项。

可知 wi+si>max⁡(si, wi+1+si+1)w_i + s_i > \max(s_i,\ w_{i+1} + s_{i+1})wi​+si​>max(si​, wi+1​+si+1​)，每次交换都不会使最大危险系数增大，因此贪心解即为最优解。

题目要求求 min⁡(最大危险系数)\min(\text{最大危险系数})min(最大危险系数)。

#include <iostream>
#include <algorithm>
using namespace std;

typedef pair<int, int> PII;

const int N = 50010;

int n;
PII cow[N];

int main(void)
{
    scanf("%d", &n);
    for (int i = 0; i < n; i++)
    {
        int s, w;
        scanf("%d%d", &w, &s);
        cow[i] = {w + s, w};
    }

    sort(cow, cow + n);

    int res = -2e9, sum = 0;
    for (int i = 0; i < n; i++)
    {
        int s = cow[i].first - cow[i].second, w = cow[i].second;
        res = max(res, sum - s);
        sum += w;
    }

    printf("%d\n", res);

    return 0;
}
