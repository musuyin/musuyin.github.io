质数

定义：在大于 1 的整数中，约数只含 1 和本身的数称为质数（素数）。

质数的判定 O(n^1/2)

试除法：逐一枚举可能的因子进行整除检验。

bool is_prime(int n)
{
	if (n == 1) return false;
    if (n == 2) return true;
    for (int i = 2; i <= n / i; i++)  //sqrt(n)太慢，i * i < n存在溢出风险
    {
        if (n % i == 0)
            return false;
    }
    return true;
}

筛质数

质数定理：1∼n1 \sim n1∼n 中约有 n/ln⁡nn / \ln nn/lnn 个质数。

在 10710^7107 的范围内，线性筛法比埃氏筛法约快一倍。

埃氏筛法 O(nlog(logn))

埃氏筛法的缺点在于：一个合数可能被多个不同的质因子重复标记删除。

#include <iostream>
using namespace std;

const int N = 1000010;

int primes[N], cnt;
bool st[N];

void get_prime(int n)
{
    for (int i = 2; i <= n; i++)
    {
        if (!st[i])
        {
            primes[cnt++] = n;  //存放质数
            for (int j = i + i; j <= n; j += i) st[j] = true;
        }	//每到一个数的时候，删去数组中该数的倍数，则当前数没有被删去就代表着是质数
			//优化：只对质数进行删除倍数的操作
    }
}

int main(void)
{
    int n;
    cin >> n;

    get_prime(n);

    cout << cnt;

    return 0;
}

线性筛法

线性筛法保证每个合数只被其最小质因子筛去，从而实现线性时间复杂度。核心逻辑如下：

  1. 若 i % primes[j] == 0，则 primes[j] 一定是 i 的最小质因子，primes[j] 也一定是 primes[j] * i 的最小质因子。
  2. 若 i % primes[j] != 0，则 primes[j] 一定小于 i 的所有质因子，primes[j] 也一定是 primes[j] * i 的最小质因子。
#include <iostream>
using namespace std;

const int N = 1000010;

int primes[N], cnt;
bool st[N];

void get_prime(int n)
{
    for (int i = 2; i <= n; i++)  	//线性筛法并不是对一个质数求完它的所有倍数
    {								//而是每次i递增时，将已有的质数的i倍true掉 2 * 2; 2 * 3; 3 * 3; 2 * 4; 3 * 4
        if (!st[i]) primes[cnt++] = i;
        for (int j = 0; primes[j] <= n / i; j++)
        {
            st[primes[j] * i] = true;  //n只会被最小质因子筛掉
            if (i % primes[j] == 0) break;  //成立时：primes[j]一定是 i 的最小质因子
        }
    }
}

int main(void)
{
    int n;
    cin >> n;

    get_prime(n);

    cout << cnt;

    return 0;
}

#include <iostream>

vector<int> get_divisors(int n)
{
    vector<int> res;

    for (int i = 1; i <= n / i; i++)
    {
        if (n % i == 0)

约数个数

当上述分解式成立时，约数个数为 (a1+1)(a2+1)⋯(an+1)(a_1 + 1)(a_2 + 1) \cdots (a_n + 1)(a1​+1)(a2​+1)⋯(an​+1)。

111 的倍数有 nnn 个，222 的倍数有 n/2n/2n/2 个……

1∼n1 \sim n1∼n 中每个数的约数之和为 nlog⁡nn \log nnlogn，即平均约数个数为 log⁡n\log nlogn。

在 int 范围内，约数个数最大约为 1500。

#include <iostream>
#include <unordered_map>
#include <algorithm>
using namespace std;

typedef long long LL;

const int mod = 1e9 + 7;

int main(void)
{
    int n;
    cin >> n;

    unordered_map<int, int> primes;  //哈希表（好像和字典很像）
    while (n--)
    {
        int x;
        cin >> x;

        for (int i = 2; i <= x / i; i++)  //试除法
        {
            while (x % i == 0)
            {
                x /= i;
                primes[i]++;  //每个数的约数个数
            }
        }

        if (x > 1) primes[x]++;  //x是质数
    }

    LL res = 1;
    for (auto prime : primes) res = res * (prime.second + 1) % mod;

    cout << res << endl;

    return 0;
}

约数之和

将所有约数相加即可得到约数之和，利用分解质因数的结果逐项累乘。

#include <iostream>
#include <unordered_map>
#include <algorithm>
using namespace std;

typedef long long LL;

const int mod = 1e9 + 7;

int main(void)
{
    int n;
    cin >> n;

    unordered_map<int, int> primes;
    while (n--)
    {
        int x;
        cin >> x;

        for (int i = 2; i <= x / i; i++)  //分解质因数
        {
            while (x % i == 0)
            {
                x /= i;
                primes[i]++;
            }
        }

        if (x > 1) primes[x]++;
    }

    LL res = 1;
    for (auto prime : primes)
    {
        int p = prime.first, a = prime.second;
        LL t = 1;  // t = p * t + 1; 1 -> p + 1 -> p^2 + p + 1 -> ...
        while (a--) t = (t * p + 1) % mod;
        res = res * t % mod;
    }

    cout << res << endl;

    return 0;
}

欧拉函数

单个数求欧拉函数 O(n^1/2)

互质是指公约数只有 1 的两个整数（任何数与 1 必然互质）。

容斥原理推导欧拉函数的过程如下：

  1. 从 1∼n1 \sim n1∼n 中去掉 p1,p2,…,pkp_1, p_2, \ldots, p_kp1​,p2​,…,pk​ 的所有倍数；
  2. 加上所有 pi⋅pjp_i \cdot p_jpi​⋅pj​ 的倍数（被重复去掉两次的倍数）；
  3. 减去所有 pi⋅pj⋅pkp_i \cdot p_j \cdot p_kpi​⋅pj​⋅pk​ 的倍数；
  4. 如此交替加减……

#include <iostream>
using namespace std;

int main(void)
{
    int n;
    cin >> n;

    while (n--)
    {
        int a;
        cin >> a;

        int res = a;
        for (int i = 2; i <= a / i; i++)  //分解质因子
        {
            if (a % i == 0)
            {
                res = res / i * (i - 1);
                while (a % i == 0) a /= i;  //质因子次数
            }
        }

        if (a > 1) res = res / a * (a - 1);  //本身为质数的情况

        cout << res << '\n';
    }

    return 0;
}

多个数求欧拉函数 O(n)

利用线性筛法，在筛质数的同时顺带求出欧拉函数，核心推导如下：

  1. 当 i % pj == 0 时，有 φ(pj⋅i)=pj⋅φ(i)\varphi(p_j \cdot i) = p_j \cdot \varphi(i)φ(pj​⋅i)=pj​⋅φ(i)（pjp_jpj​ 是 iii 的一个质因子）

    • φ[i]=i⋅(1−1/p1)⋅(1−1/p2)⋯(1−1/pk)\varphi[i] = i \cdot (1 - 1/p_1) \cdot (1 - 1/p_2) \cdots (1 - 1/p_k)φ[i]=i⋅(1−1/p1​)⋅(1−1/p2​)⋯(1−1/pk​)
    • φ[i⋅pj]=(i⋅pj)⋅(1−1/p1)⋅(1−1/p2)⋯(1−1/pk)\varphi[i \cdot p_j] = (i \cdot p_j) \cdot (1 - 1/p_1) \cdot (1 - 1/p_2) \cdots (1 - 1/p_k)φ[i⋅pj​]=(i⋅pj​)⋅(1−1/p1​)⋅(1−1/p2​)⋯(1−1/pk​)
    • 由此可得 φ[i⋅pj]=pj⋅φ[i]\varphi[i \cdot p_j] = p_j \cdot \varphi[i]φ[i⋅pj​]=pj​⋅φ[i]

  2. 当 i % pj != 0 时，有 φ(pj⋅i)=(pj−1)⋅φ(i)\varphi(p_j \cdot i) = (p_j - 1) \cdot \varphi(i)φ(pj​⋅i)=(pj​−1)⋅φ(i)（pjp_jpj​ 不是 iii 的质因子）

    • φ[i]=i⋅(1−1/p1)⋅(1−1/p2)⋯(1−1/pk)\varphi[i] = i \cdot (1 - 1/p_1) \cdot (1 - 1/p_2) \cdots (1 - 1/p_k)φ[i]=i⋅(1−1/p1​)⋅(1−1/p2​)⋯(1−1/pk​)
    • φ[i⋅pj]=(i⋅pj)⋅(1−1/p1)⋅(1−1/p2)⋯(1−1/pk)⋅(1−1/pj)\varphi[i \cdot p_j] = (i \cdot p_j) \cdot (1 - 1/p_1) \cdot (1 - 1/p_2) \cdots (1 - 1/p_k) \cdot (1 - 1/p_j)φ[i⋅pj​]=(i⋅pj​)⋅(1−1/p1​)⋅(1−1/p2​)⋯(1−1/pk​)⋅(1−1/pj​)
    • 由此可得 φ[i⋅pj]=φ[i]⋅(pj−1)\varphi[i \cdot p_j] = \varphi[i] \cdot (p_j - 1)φ[i⋅pj​]=φ[i]⋅(pj​−1)
#include <iostream>
#include <algorithm>
using namespace std;

const int N = 1000010;
typedef long long LL;
int primes[N], cnt;
int euler[N];
bool st[N];

LL get_eulers(int n)
{
    euler[1] = 1;
    for (int i = 2; i <= n; i++)
    {
        if (!st[i])
        {
            primes[cnt++] = i;
            euler[i] = i - 1;  //质数n本身有n - 1个数与其互质（费马小定理）
        }
						//primes[j] * i <= n
        for (int j = 0; primes[j] <= n / i; j++)  //线性筛法(将这个质数的所有的倍数的互质数求出来)
        {
            st[primes[j] * i] = true;  //i * prime[j]的最小质因子就是prime[j]，则prime[j] * i就不是质数了，把它true掉
            if (i % primes[j] == 0)  //这个质因子可以把i整除完了，说明其后再没有i的质因子了
            {
                euler[primes[j] * i] = euler[i] * primes[j];
                break;
            }
            euler[primes[j] * i] = euler[i] * (primes[j] - 1);
        }
    }

    LL res = 0;
    for (int i = 1; i <= n; i++) res += euler[i];
    return res;
}

int main(void)
{
    int n;
    cin >> n;
    cout << get_eulers(n) << '\n';

    return 0;
}

快速幂

快速幂

快速求出 ak mod pa^k \bmod pakmodp，时间复杂度为 O(log⁡k)O(\log k)O(logk)。

核心思路：反复平方法。

先预处理出：

前一项直接平方后取模即可得到下一项。

将 kkk 拆成二进制，按位累乘对应的幂次。

#include <iostream>
#include <algorithm>
using namespace std;

typedef long long LL;

int qmi(int a, int k, int p)
{
    int res = 1;
    while (k)
    {
        if (k & 1) res = (LL)res * a % p;  //LL防止数据溢出（大部分数论题都需要long long）
        k >>= 1;
        a = (LL)a * a % p;
    }
    return res;
}

int main(void)
{
    int n;
    scanf("%d", &n);
    while (n--)
    {
        int a, k, p;
        scanf("%d %d %d", &a, &k, &p);

        printf("%d\n", qmi(a, k, p));
    }

    return 0;
}

快速幂求逆元

乘法逆元的意义在于将除运算转换为等价的乘运算，从而在模运算下处理除法。

由定义 b⋅b−1≡1(modm)b \cdot b^{-1} \equiv 1 \pmod{m}b⋅b−1≡1(modm)，再结合费马小定理 bm−1≡1(modm)b^{m-1} \equiv 1 \pmod{m}bm−1≡1(modm)（mmm 为质数），可得：

b⋅bm−2≡1(modm)b \cdot b^{m-2} \equiv 1 \pmod{m}b⋅bm−2≡1(modm)

因此 b−1≡bm−2(modm)b^{-1} \equiv b^{m-2} \pmod{m}b−1≡bm−2(modm)，用快速幂即可求解。

注意：当 bbb 为 mmm 的倍数时，b mod m=0b \bmod m = 0bmodm=0，逆元不存在。

#include <iostream>
using namespace std;

typedef long long LL;

int qmi(int a, int k, int p)
{
    int res = 1;
    while (k)
    {
        if (k & 1) res = (LL)res * a % p;
        k >>= 1;
        a = (LL)a * a % p;
    }

    return res;
}

int main(void)
{
    int n;
    scanf("%d", &n);
    while (n--)
    {
        int a, p;
        scanf("%d %d", &a, &p);  //必须确保p是质数才能算逆元
        //判断a是否为p的倍数
        if (a % p) printf("%d\n", qmi(a, p - 2, p));  //公式有费马小定理可以推出
        else puts("impossible");
    }

    return 0;
}

裴蜀定理

对于任意正整数 aaa、bbb，一定存在整数 xxx、yyy，使得 ax+by=gcd⁡(a,b)ax + by = \gcd(a, b)ax+by=gcd(a,b)。

推理：aaa 和 bbb 均为 gcd⁡(a,b)\gcd(a, b)gcd(a,b) 的倍数，因此 ax+byax + byax+by 也是 gcd⁡(a,b)\gcd(a, b)gcd(a,b) 的倍数；取该系数使倍数恰好为 1 即证。

扩展欧几里得算法

扩展欧几里得算法用于求裴蜀定理中的具体系数 xxx、yyy。

#include <iostream>

int exgcd(int a, int b, int & x, int & y)
{
    if (!b)
    {
        x = 1, y = 0;
        return a;
    }

    int d = exgcd(b, a % b, y, x);

}

高斯消元

高斯消元解线性方程组

nnn 元线性方程组的解有三种情况：

  1. 无解
  2. 无穷多解
  3. 唯一解

正方形矩阵的初等行变换（不影响方程组的解）：

  1. 将某一行乘以一个非零的数；
  2. 交换某两行；
  3. 将某行的若干倍加到另一行。

通过初等行变换将方程组化为上三角形式：

  1. 完美阶梯形：唯一解；
  2. 出现 0=非零0 = \text{非零}0=非零：无解；
  3. 出现 0=00 = 00=0：无穷多解。

高斯消元步骤：

  1. 枚举每一列（第 ccc 列）：
    1. 找到该列绝对值最大的行；
    2. 将该行换到当前处理行的最上面；
    3. 将该行首位系数化为 1（整行除以首位系数）；
    4. 将下方所有行的第 ccc 列消为 0。
  2. 得到上三角矩阵后，从最下层开始回代求解，逐步代入上层方程，直到所有未知数求出。
#include <iostream>
#include <cstring>
#include <algorithm>
#include <cmath>
using namespace std;

const int N = 110;
const double eps = 1e-8;

int n;
double a[N][N];

int gauss(void)  //答案存于a[i][n]中
{
    int c, r;
    for (c = 0, r = 0; c < n; c++)
    {
        int t = r;
        for (int i = r; i < n; i++)  //找绝对值最大的一行
            if (fabs(a[i][c] > fabs(a[t][c])))
                t = i;

            if (fabs(a[t][c]) < eps) continue;  //这一列的系数全为0

            for (int i = c; i <= n; i++) swap(a[t][i], a[r][i]);  //将最大值换到最顶端
            for (int i = n; i >= c; i--) a[r][i] /= a[r][c];  //将当前行首位变成1
            for (int i = r + 1; i < n; i++)  //将下面的所有列消成0
            {
                if (fabs(a[i][c]) > eps)
                    for (int j = n; j >= c; j--)
                        a[i][j] -= a[r][j] * a[i][c];  //同时，每一行的其他系数也会发生改变
            }

            r++;  //下一列
    }
        //迭代次数和列数不相同，则说明无唯一解
        if (r < n)
        {
            for (int i = r; i < n; i++)
                if (fabs(a[i][n]) > eps)  //存在一行一侧无法被消成0
                    return 2;  //无解
            return 1;
        }

        //从底下往上求出方程的解
        for (int i = n - 1; i >= 0; i--)
            for (int j = i + 1; j < n; j++)
                a[i][n] -= a[i][j] * a[j][n];

    return 0;  //有唯一解
}

int main(void)
{
    scanf("%d", &n);
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n + 1; j++)
            scanf("%lf", &a[i][j]);

    int t = gauss();
    if (t == 2) puts("No solution");
    else if (t == 1) puts("Infinite group solutions");
    else
    {
        for (int i = 0; i < n; i++)
            printf("%.2f\n", a[i][n]);
    }

    return 0;
}

高斯消元解异或方程组

异或运算等价于不进位的加法，因此可将其视为 GF(2)\text{GF}(2)GF(2) 上的线性方程组进行高斯消元。

  1. 消成上三角矩阵后：
    1. 有效行数等于 nnn：唯一解；
    2. 有效行数不等于 nnn，且存在矛盾行：无解；
    3. 有效行数不等于 nnn，且无矛盾行：无穷多解。
#include <iostream>
#include <algorithm>
using namespace std;

const int N = 110;

int n;
int a[N][N];

int gauss(void)
{
    int c, r;
    for (c = 0, r = 0; c < n; c++)
    {
        int t = r;
        for (int i = r; i < n; i++)
            if (a[i][c])
                t = i;

        if (!a[t][c]) continue;

        for (int i = c; i <= n; i++) swap(a[r][i], a[t][i]);
        for (int i = r + 1; i < n; i++)
            if (a[i][c])
                for (int j = n; j >= c; j--)
                    a[i][j] ^= a[r][j];

        r++;
    }

    if (r < n)  //异常
    {
        for (int i = r; i < n; i++)
            if (a[i][n])
                return 2;
        return 1;
    }

    for (int i = n - 1; i >= 0; i--)
        for (int j = i + 1; j < n; j++)
            a[i][n] ^= a[i][j] & a[j][n];  //对于单个位来说，^ = +, & = *

    return 0;
}

int main(void)
{
    cin >> n;

    for (int i = 0; i < n; i++)
        for (int j = 0; j < n + 1; j++)
            cin >> a[i][j];

    int t = gauss();

    if (t == 0)
        for (int i = 0; i < n; i++)
            cout << a[i][n] << '\n';
    else if (t == 1)
        puts("Multiple sets of solutions");
    else
        puts("No solution");

    return 0;
}

{
    if (b > a) return 0;

    int res = 1;
    for (int i = 1, j = a; i <= b; i++, j--)
    {

高精度计算

将组合数进行质因数分解，转化为只含乘法的形式，再用高精度乘法输出结果。

一个数含一个 ppp 计一次，含 kkk 个 ppp 计 kkk 次。

步骤：

  1. 筛质数；
  2. 求各质因子的次数；
  3. 高精度乘法累乘。
#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

const int N = 5010;
typedef vector<int> VI;

int primes[N], cnt;
int sum[N];
bool st[N];

void get_primes(int n)  //线性筛
{
    for (int i = 2; i <= n; i++)
    {
        if (!st[i]) primes[cnt++] = i;
        for (int j = 0; primes[j] <= n / i; j++)
        {
            st[primes[j] * i] = true;
            if (i % primes[j] == 0) break;
        }
    }
}

int get(int n, int p)
{
    int res = 0;
    while (n)
    {
        res += n / p;
        n /= p;
    }
    return res;
}

VI mul(VI a, int b)  //高精度乘法
{
    VI c;
    int t = 0;
    for (int i = 0; i < a.size(); i++)
    {
        t += a[i] * b;
        c.push_back(t % 10);
        t /= 10;
    }
    while (t)
    {
        c.push_back(t % 10);
        t /= 10;
    }
    return c;
}

int main(void)
{
    int a, b;
    cin >> a >> b;

    get_primes(a);

    for (int i = 0; i < cnt; i++)
    {   //分解质因数
        int p = primes[i];  //分别以每个质数为底进行
        sum[i] = get(a, p) - get(a - b, p) - get(b, p);  //分子次数-分母次数
    }

    VI res;
    res.push_back(1);  //初始为1

    for (int i = 0; i < cnt; i++)
        for (int j = 0; j < sum[i]; j++)
            res = mul(res, primes[i]);

    for (int i = res.size() - 1; i >= 0; i--) printf("%d", res[i]);
    puts("");

    return 0;
}

卡特兰数

满足条件的 01 序列

每一个数列都对应一条格路径。

在任意时刻，需要满足 x≥yx \geq yx≥y。

答案数 === 总路线数 −-− 经过红色线的路线数。

当一条路径穿过了红线，将该路径与红线的交点之后的部分关于红线作对称，则终点必然落在 (5,7)(5, 7)(5,7)（即终点 (6,6)(6, 6)(6,6) 关于红线的对称点）。

因此，所有能走到 (5,7)(5, 7)(5,7) 的路径，必然有且仅有一条对应的穿过红线且到达 (6,6)(6, 6)(6,6) 的路径与之对应。

可得：每一条从 (0,0)(0, 0)(0,0) 到 (6,6)(6, 6)(6,6) 且穿过红线的路径，均可唯一表示为从 (0,0)(0, 0)(0,0) 到 (5,7)(5, 7)(5,7) 的路径。

最终答案：

由于每个 ppp 均为质数，kkk 个质数乘积 pi⋯pjp_i \cdots p_jpi​⋯pj​ 的倍数有 ⌊n/(pi⋯pj)⌋\lfloor n / (p_i \cdots p_j) \rfloor⌊n/(pi​⋯pj​)⌋ 个（无需求最大公约数）。

使用二进制枚举所有子集的选法，以所选质数个数的奇偶性判断加减：奇数个为正，偶数个为负。

#include <iostream>
#include <algorithm>
using namespace std;

typedef long long LL;
const int N = 20;
int p[N];

int main(void)
{
    int n, m;
    cin >> n >> m;

    for (int i = 0; i < m; i++) cin >> p[i];

    int res = 0;
    for (int i = 1; i < 1 << m; i++)
    {
        int t = 1, s = 0;
        for (int j = 0; j < m; j++)
            if (i >> j & 1)  //如果为1代表选了
            {
                if ((LL)t * p[j] > n)  //所需要相乘的质数的积已经大于n了
                {
                    t = -1;
                    break;
                }
                t *= p[j];
                s++;
            }

        if (t != -1)
        {
            if (s % 2) res += n / t;  //s为奇数
            else res -= n / t;
        }
    }

    cout << res << '\n';

    return 0;
}

#include <cstring>
#include <iostream>
#include <algorithm>
#include <unordered_set>
using namespace std;

const int N = 110, M = 10010;

int n, m;
int s[N], f[M];  //f记忆化存储

int sg(int x)
{
    if (f[x] != -1) return f[x]; //记忆化存储

    unordered_set<int> S;  //hash table
    for (int i = 0; i < m; i++)
    {
        int sum = s[i];  //一次拿的数量
        if (x >= sum) S.insert(sg(x - sum));  //如果石子数不少于拿的数量
        //将x - sum的情况存入S中（考虑图示，只需要将相邻的SG值存入就可以了）
    }

    for (int i = 0; ; i++)  //找到不在S中的最小自然数
        if (!S.count(i))  //在S中寻找i，找不到就赋值（count()返回某一个数的个数）
            return f[x] = i;
}

    //如果x不同，那么也不会相互干扰
