ACM 输入输出

2026-10-28

#ACM 模式#Java#Go

阅读线索

ACM 输入输出按题目协议处理数据。Java 与 Go 都应先确定按词、按行还是读到 EOF，再选读取器；缓冲输出和大行输入还有各自的边界。

ACM 模式

核心代码模式只要求实现函数，评测平台负责构造参数和接收返回值。ACM 模式要求程序自行从标准输入读取数据，并把答案写入标准输出。

输入通常按空白字符切分；只有题目明确要求保留空格或读取整行文本时，才按行读取。

Java

Scanner

适合数据量较小、以代码简洁为主的题目。next() 按空白字符读取，nextLine() 读取一整行；两者混用时要注意上一条输入留下的换行符。

static void scannerExample() {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    long value = scanner.nextLong();
    String word = scanner.next();
    System.out.println(n + " " + value + " " + word);
}

BufferedReader

需要读取整行，或希望自行控制分割规则时使用。

static int[] readIntLine(BufferedReader reader) throws IOException {
    String[] fields = reader.readLine().trim().split("\\s+");
    int[] values = new int[fields.length];
    for (int i = 0; i < fields.length; i++) {
        values[i] = Integer.parseInt(fields[i]);
    }
    return values;
}

快速输入

大量整数输入时，直接读取字节并跳过空白字符。题目保证输入合法，因此解析器不额外吞掉格式错误。

static class FastScanner {
    private final InputStream input = System.in;
    private final byte[] buffer = new byte[1 << 16];
    private int pointer;
    private int length;

    private int read() throws IOException {
        if (pointer >= length) {
            length = input.read(buffer);
            pointer = 0;
            if (length == -1) return -1;
        }
        return buffer[pointer++];
    }

    long nextLong() throws IOException {
        int c;
        do {
            c = read();
        } while (c <= ' ');
        int sign = 1;
        if (c == '-') {
            sign = -1;
            c = read();
        }
        long value = 0;
        while (c > ' ') {
            value = value * 10 + c - '0';
            c = read();
        }
        return sign * value;
    }

    int nextInt() throws IOException {
        return Math.toIntExact(nextLong());
    }
}

缓冲输出

BufferedWriter 的内容必须经过 flush() 才能保证写入标准输出。批量拼接答案时也可以使用 StringBuilder，最后一次性输出。

static void bufferedOutput(int[] values) throws IOException {
    BufferedWriter writer = new BufferedWriter(new OutputStreamWriter(System.out));
    for (int i = 0; i < values.length; i++) {
        if (i > 0) writer.write(" ");
        writer.write(Integer.toString(values[i]));
    }
    writer.newLine();
    writer.flush();
}

Go

Go 公共定义

以下不含 package 的函数模板共享这组输入输出对象；完整 ACM 程序各自独立使用。

import (
	"bufio"
	"os"
)

var in = bufio.NewReader(os.Stdin)
var out = bufio.NewWriter(os.Stdout)

fmt.Scan

适合数据量较小的简单输入，按空白字符读取。

package main

import "fmt"

func main() {
	var a, b int
	fmt.Scan(&a, &b)
	fmt.Println(a + b)
}

缓冲输入输出

ACM 模式优先使用 bufio.Reader 与 bufio.Writer。fmt.Fscan 仍按空白字符切分，但避免每次直接访问标准输入输出。

package main

import (
	"bufio"
	"fmt"
	"os"
)

var in = bufio.NewReader(os.Stdin)
var out = bufio.NewWriter(os.Stdout)

func main() {
	defer out.Flush()

	var n int
	fmt.Fscan(in, &n)
	nums := make([]int, n)
	for i := range nums {
		fmt.Fscan(in, &nums[i])
	}

	for i, value := range nums {
		if i > 0 {
			fmt.Fprint(out, " ")
		}
		fmt.Fprint(out, value)
	}
	fmt.Fprintln(out)
}

读取到 EOF

正常 EOF 表示读取完成；其他解析错误应显式报错，避免把非法输入当成结束。

import (
	"fmt"
	"io"
)

func readUntilEOF() {
	defer out.Flush()
	for {
		var value int
		if _, err := fmt.Fscan(in, &value); err == io.EOF {
			break
		} else if err != nil {
			panic(err)
		}
		fmt.Fprintln(out, value)
	}
}

读取整行

Scanner 默认按行切分，单行可能超过 64 KiB 时必须扩大缓冲区。

import (
	"bufio"
	"fmt"
	"os"
)

func readLines() {
	defer out.Flush()
	scanner := bufio.NewScanner(os.Stdin)
	scanner.Buffer(make([]byte, 1024), 1024*1024)
	for scanner.Scan() {
		fmt.Fprintln(out, scanner.Text())
	}
	if err := scanner.Err(); err != nil {
		panic(err)
	}
}

固定格式与单行扫描

Scanf 按格式读取；Scanln 在换行处停止，要求本行提供全部待填变量。对应的 Fscanf、Fscanln 接收同一个 Reader。

import "fmt"

func readFormatted() (int, int) {
	var a, b int
	fmt.Fscanf(in, "%d,%d", &a, &b)
	return a, b
}

func readSingleLine() (int, int) {
	var a, b int
	fmt.Fscanln(in, &a, &b)
	return a, b
}

固定组数与矩阵

import "fmt"

func readFixedCases() {
	defer out.Flush()
	var count int
	fmt.Fscan(in, &count)
	for i := 0; i < count; i++ {
		var a, b int
		fmt.Fscan(in, &a, &b)
		fmt.Fprintln(out, a+b)
	}
}

func readMatrix() [][]int {
	var rows, columns int
	fmt.Fscan(in, &rows, &columns)
	grid := make([][]int, rows)
	for i := range grid {
		grid[i] = make([]int, columns)
		for j := range grid[i] {
			fmt.Fscan(in, &grid[i][j])
		}
	}
	return grid
}

单行整数切片

对整行用 strings.Fields 切分，再逐个转换；转换失败时保留错误。函数接收 Scanner，同一输入流始终使用同一 Scanner，不与其他缓冲读取器交替读取。

import (
	"bufio"
	"strconv"
	"strings"
)

func readIntegerLine(scanner *bufio.Scanner) []int {
	if !scanner.Scan() {
		if err := scanner.Err(); err != nil {
			panic(err)
		}
		panic("缺少预期的整数行")
	}
	fields := strings.Fields(scanner.Text())
	values := make([]int, len(fields))
	for i, field := range fields {
		value, err := strconv.Atoi(field)
		if err != nil {
			panic(err)
		}
		values[i] = value
	}
	return values
}

格式化与空格分隔输出

Fprint 不自动换行，Fprintln 输出后换行，Fprintf 使用格式字符串。整数切片也可转成字符串切片后用 strings.Join 输出，末尾没有多余空格。

import (
	"fmt"
	"strconv"
	"strings"
)

func writeIntegers(values []int) {
	defer out.Flush()
	fields := make([]string, len(values))
	for i, value := range values {
		fields[i] = strconv.Itoa(value)
	}
	fmt.Fprintln(out, strings.Join(fields, " "))
}

func writeFormatted(value int) {
	defer out.Flush()
	fmt.Fprintf(out, "%d\n", value)
}
