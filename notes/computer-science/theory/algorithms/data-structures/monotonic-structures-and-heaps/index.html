单调结构与堆

2026-10-28

#单调栈#单调队列#堆

阅读线索

单调结构保留仍可能成为答案的候选，堆只保证堆顶极值。正确性来自被弹出元素不再有用，均摊复杂度则来自每个元素至多进出一次。

单调栈与单调队列通过及时删除不可能再成为答案的元素，把重复比较降为均摊 O(1)O(1)O(1)。堆只保证堆顶最优，适合动态维护极值。

左侧最近的较小元素

给定整数序列，对每个元素输出其左侧离它最近且严格小于它的元素；不存在时输出 −1-1−1。

思路

维护从栈底到栈顶严格递增的值。处理 xxx 时，所有大于等于 xxx 的栈顶元素不可能成为 xxx 或未来更优的候选，先弹出；剩余栈顶就是答案。

int[] previousSmaller(int[] nums) {
    int[] answer = new int[nums.length];
    Deque<Integer> stack = new ArrayDeque<>();
    for (int i = 0; i < nums.length; i++) {
        while (!stack.isEmpty() && stack.peek() >= nums[i]) {
            stack.pop();
        }
        answer[i] = stack.isEmpty() ? -1 : stack.peek();
        stack.push(nums[i]);
    }
    return answer;
}

直方图最大矩形

给定每根柱子的高度，柱宽均为 111，求直方图中能够形成的最大矩形面积。

思路

维护柱高严格递增的下标栈。遇到更矮的柱子时，弹出的柱子以当前下标为第一个右侧更矮位置，弹出后的新栈顶为第一个左侧更矮位置，据此确定它作为最矮柱时的最大宽度。两端补高度为 000 的哨兵统一边界。

int largestRectangleArea(int[] heights) {
    int[] extended = new int[heights.length + 2];
    System.arraycopy(heights, 0, extended, 1, heights.length);
    Deque<Integer> stack = new ArrayDeque<>();
    stack.push(0);
    int answer = 0;
    for (int i = 1; i < extended.length; i++) {
        while (extended[stack.peek()] > extended[i]) {
            int height = extended[stack.pop()];
            int width = i - stack.peek() - 1;
            answer = Math.max(answer, height * width);
        }
        stack.push(i);
    }
    return answer;
}

滑动窗口最值

给定长度为 nnn 的数组和窗口大小 kkk，求每个滑动窗口中的最大值。

思路

双端队列保存候选下标，并保证对应值从队首到队尾递减：

  • 队首超出窗口左边界时删除。
  • 新值不小于队尾值时，队尾以后不可能成为最大值，持续删除。
  • 当前队首就是窗口最大值。
int[] maxSlidingWindow(int[] nums, int k) {
    int[] answer = new int[nums.length - k + 1];
    Deque<Integer> deque = new ArrayDeque<>();
    for (int i = 0; i < nums.length; i++) {
        while (!deque.isEmpty() && deque.peekFirst() <= i - k) {
            deque.pollFirst();
        }
        while (!deque.isEmpty() && nums[deque.peekLast()] <= nums[i]) {
            deque.pollLast();
        }
        deque.offerLast(i);
        if (i >= k - 1) {
            answer[i - k + 1] = nums[deque.peekFirst()];
        }
    }
    return answer;
}

Go 实现：滑动窗口最大值

type node struct {
	idx int
	val int
}

func (this *node) less(x node) bool {
	return this.val < x.val
}

type MonoDescQueue struct {
	arr  []*node
	size int
}

func NewMonoDescQueue() MonoDescQueue {
	return MonoDescQueue{
		arr:  []*node{},
		size: 0,
	}
}

func (this *MonoDescQueue) Push(x *node) {
	for this.size > 0 && this.arr[this.size-1].less(*x) {
		this.arr = this.arr[:this.size-1]
		this.size--
	}
	this.arr = append(this.arr, x)
	this.size++
}

func (this *MonoDescQueue) Pop() *node {
	ret := this.arr[0]
	this.arr = this.arr[1:]
	this.size--
	return ret
}

func (this *MonoDescQueue) Top() *node {
	return this.arr[0]
}

func maxSlidingWindow(nums []int, k int) []int {
	size := len(nums)
	queue := NewMonoDescQueue()
	ret := make([]int, size-k+1)
	for i, num := range nums {
		// insert into queue
		queue.Push(&node{i, num})
		// do not judge the first k node
		if i < k-1 {
			continue
		}

		// check whether the first node in queue is expired
		// only one or zero node will be removed every round
		top := queue.Top()
		if i-k >= top.idx {
			queue.Pop()
		}
		// update the current biggest node
		top = queue.Top()
		ret[i-k+1] = top.val
	}
	return ret
}

小根堆

完全二叉树用数组存储时，下标从 111 开始最方便：节点 uuu 的左右孩子分别为 2u2u2u 和 2u+12u+12u+1。建堆从最后一个非叶节点向前执行 down，复杂度为 O(n)O(n)O(n)。

void heapSort(int[] nums) {
    int[] heap = new int[nums.length + 1];
    for (int i = 0; i < nums.length; i++) {
        heap[i + 1] = nums[i];
    }
    int size = nums.length;
    for (int i = size / 2; i >= 1; i--) {
        down(heap, i, size);
    }
    for (int i = 0; i < nums.length; i++) {
        nums[i] = heap[1];
        heap[1] = heap[size--];
        down(heap, 1, size);
    }
}

void down(int[] heap, int index, int size) {
    while (index * 2 <= size) {
        int child = index * 2;
        if (child + 1 <= size && heap[child + 1] < heap[child]) {
            child++;
        }
        if (heap[index] <= heap[child]) {
            return;
        }
        int temporary = heap[index];
        heap[index] = heap[child];
        heap[child] = temporary;
        index = child;
    }
}

前若干个高频元素

给定整数数组和整数 kkk，返回出现频率最高的 kkk 个元素。

思路

先用哈希表计数，再维护容量为 kkk 的小根堆。堆顶是当前候选中频率最低的元素，超过容量时删除堆顶。复杂度为 O(nlog⁡k)O(n \log k)O(nlogk)。

int[] topKFrequent(int[] nums, int k) {
    Map<Integer, Integer> frequency = new HashMap<>();
    for (int x : nums) {
        frequency.merge(x, 1, Integer::sum);
    }
    PriorityQueue<Integer> heap = new PriorityQueue<>(
            Comparator.comparingInt(frequency::get));
    for (int x : frequency.keySet()) {
        heap.offer(x);
        if (heap.size() > k) {
            heap.poll();
        }
    }
    int[] answer = new int[k];
    for (int i = k - 1; i >= 0; i--) {
        answer[i] = heap.remove();
    }
    return answer;
}

Go 实现：前 K 个高频元素

Go 将全部不同元素放入大根堆再弹出 kkk 个。若不同元素数为 uuu，时间为 O(n+ulog⁡u+klog⁡u)O(n+u\log u+k\log u)O(n+ulogu+klogu)，空间为 O(u)O(u)O(u)。Go 标准库 container/heap 也支持优先队列，这里保留手写堆。

type node struct {
	val   int
	times int
}

func (this *node) less(x *node) bool {
	return this.times < x.times
}

type PriorityQueue struct {
	arr      []*node
	size     int
	capacity int
}

func NewPriorityQueue(capacity int) PriorityQueue {
	return PriorityQueue{
		arr:      make([]*node, capacity+1),
		size:     0,
		capacity: capacity,
	}
}

func (this *PriorityQueue) Push(x *node) {
	this.size++
	this.arr[this.size] = x
	this.swim()
}

func (this *PriorityQueue) Pop() *node {
	ret := this.arr[1]
	this.swap(1, this.size)
	this.size--
	this.sink(1)
	return ret
}

func (this *PriorityQueue) Top() *node {
	return this.arr[1]
}

func (this *PriorityQueue) swim() {
	cur := this.size
	for cur/2 > 0 && this.arr[cur/2].less(this.arr[cur]) {
		this.swap(cur, cur/2)
		cur = cur / 2
	}
}

func (this *PriorityQueue) sink(x int) {
	cur := x
	nxt := cur * 2
	if nxt <= this.size && this.arr[cur].less(this.arr[nxt]) {
		cur = nxt
	}
	if nxt+1 <= this.size && this.arr[cur].less(this.arr[nxt+1]) {
		cur = nxt + 1
	}
	if cur != x {
		this.swap(x, cur)
		this.sink(cur)
	}
}

func (this *PriorityQueue) swap(a, b int) {
	this.arr[a], this.arr[b] = this.arr[b], this.arr[a]
}

func topKFrequent(nums []int, k int) []int {
	times := make(map[int]int)
	for _, num := range nums {
		times[num]++
	}
	pq := NewPriorityQueue(len(times))
	for val, times := range times {
		node := &node{
			val:   val,
			times: times,
		}
		pq.Push(node)
	}

	ret := make([]int, k)
	for i := 0; i < k; i++ {
		ret[i] = pq.Pop().val
	}
	return ret
}

最大空闲矩形

给定只包含可用与不可用单元格的矩阵，求全部由可用单元格构成的最大轴对齐矩形面积。

思路

逐行更新每一列连续可用单元格的高度。固定当前行为矩形下边界后，问题就是当前高度数组上的“直方图最大矩形”。

复杂度

  • 时间复杂度：O(rc)O(rc)O(rc)。
  • 空间复杂度：O(c)O(c)O(c)。
int largestFreeRectangle(boolean[][] available) {
    int columns = available[0].length;
    int[] heights = new int[columns];
    int answer = 0;
    for (boolean[] row : available) {
        for (int column = 0; column < columns; column++) {
            heights[column] = row[column] ? heights[column] + 1 : 0;
        }
        answer = Math.max(answer, largestRectangleArea(heights));
    }
    return answer;
}

二维滑动窗口最值

给定 n×mn\times mn×m 矩阵和窗口大小 a×ba\times ba×b，对每个窗口求最大值与最小值，并累加二者乘积。

思路

可分离地执行两次一维单调队列：

  1. 对每一行求所有宽为 bbb 的窗口最大值与最小值。
  2. 对每个横向窗口起点，再沿列方向求高为 aaa 的窗口最大值与最小值。

复杂度

  • 时间复杂度：O(nm)O(nm)O(nm)。
  • 空间复杂度：O(nm)O(nm)O(nm)。
long sumWindowExtremaProducts(int[][] matrix, int height, int width, int mod) {
    int rows = matrix.length;
    int columns = matrix[0].length;
    int horizontalCount = columns - width + 1;
    int[][] rowMaximum = new int[rows][horizontalCount];
    int[][] rowMinimum = new int[rows][horizontalCount];
    for (int row = 0; row < rows; row++) {
        rowMaximum[row] = slidingMaximum(matrix[row], width);
        rowMinimum[row] = slidingMinimum(matrix[row], width);
    }
    long answer = 0;
    int[] columnMaximum = new int[rows];
    int[] columnMinimum = new int[rows];
    for (int column = 0; column < horizontalCount; column++) {
        for (int row = 0; row < rows; row++) {
            columnMaximum[row] = rowMaximum[row][column];
            columnMinimum[row] = rowMinimum[row][column];
        }
        int[] maximum = slidingMaximum(columnMaximum, height);
        int[] minimum = slidingMinimum(columnMinimum, height);
        for (int row = 0; row < maximum.length; row++) {
            answer = (answer + (long) maximum[row] * minimum[row]) % mod;
        }
    }
    return answer;
}

int[] slidingMaximum(int[] values, int window) {
    int[] answer = new int[values.length - window + 1];
    Deque<Integer> deque = new ArrayDeque<>();
    for (int i = 0; i < values.length; i++) {
        while (!deque.isEmpty() && deque.peekFirst() <= i - window) deque.removeFirst();
        while (!deque.isEmpty() && values[deque.peekLast()] <= values[i]) deque.removeLast();
        deque.addLast(i);
        if (i >= window - 1) answer[i - window + 1] = values[deque.peekFirst()];
    }
    return answer;
}

int[] slidingMinimum(int[] values, int window) {
    int[] answer = new int[values.length - window + 1];
    Deque<Integer> deque = new ArrayDeque<>();
    for (int i = 0; i < values.length; i++) {
        while (!deque.isEmpty() && deque.peekFirst() <= i - window) deque.removeFirst();
        while (!deque.isEmpty() && values[deque.peekLast()] >= values[i]) deque.removeLast();
        deque.addLast(i);
        if (i >= window - 1) answer[i - window + 1] = values[deque.peekFirst()];
    }
    return answer;
}
