Gorm

2026-07-24

#Go#Web

GORM 用结构体和链式 API 表达数据库操作，但连接池、事务边界、迁移与 SQL 安全仍由应用负责。先确认 DSN 和连接，再理解模型映射，最后逐个验证 CRUD 的实际 SQL 与影响行数。

安装

go get -u gorm.io/gorm
go get -u gorm.io/driver/mysql

连接数据库

使用 GORM 连接数据库时，需要提供数据源名称：

  • 用户名
  • 密码
  • 主机地址
  • 端口号
  • 数据库名
package main

import (
    "gorm.io/driver/mysql"
    "gorm.io/gorm"
    "log"
)

func main() {
    // 定义数据源名称
    dsn := "root:123456@tcp(127.0.0.1:3306)/agenthub?charset=utf8mb4&parseTime=True&loc=Local"

    // 连接数据库
    db, err := gorm.Open(mysql.Open(dsn), &gorm.Config{})
    if err != nil {
        log.Fatalf("failed to connect database: %v", err)
    }
}

初始化连接

GORM 底层使用 database/sql 连接池；应用应按数据库容量与负载配置最大连接数、空闲连接数和连接寿命。见 连接数据库

sqlDB, err := db.DB()
if err != nil {
	log.Fatalf("failed to get database object: %v", err)
}
// 设置最大打开连接数
sqlDB.SetMaxOpenConns(100)

// 设置最大空闲连接数
sqlDB.SetMaxIdleConns(10)

定义模型

通过结构体来映射数据库表，每个字段对应表的一列

type Student struct {
	ID   uint   `gorm:"primaryKey;autoIncrement"`
	Name string `gorm:"size:255"`
	Age  int    `gorm:"type:int"`
}

让 GORM 根据 Go 类型自动推断 MySQL 类型（string → varchar(255)，int → bigint，uint → bigint unsigned），不需要手动指定 type，反而更稳。

如果确实需要精确控制类型，用 type:varchar(255) 把长度直接写在 type 里

自动迁移

根据结构体生成数据库表

可以根据模型结构自动创建或者更新数据库表

db.AutoMigrate(&Student{})

GORM Gen gormt
来源 官方 社区
输出 结构体 + 类型安全查询 仅结构体
适用场景 新项目、想要完整 Gen 生态 只需快速拿到结构体

gorm.io/gen

gorm.io/gen 是 GORM 官方提供的代码生成工具，可以连接数据库反向生成结构体和类型安全的查询代码

go get gorm.io/gen

生成脚本

package main

import (
    "gorm.io/driver/mysql"
    "gorm.io/gorm"
    "gorm.io/gen"
)

func main() {
    db, _ := gorm.Open(mysql.Open("user:pass@tcp(127.0.0.1:3306)/dbname"))

    g := gen.NewGenerator(gen.Config{
        OutPath: "./dal/query",     // 查询代码输出目录
        ModelPkgPath: "./dal/model", // 结构体输出目录
        Mode: gen.WithDefaultQuery | gen.WithQueryInterface,
    })

    g.UseDB(db)

    // 指定表，或用 g.GenerateAllTable() 生成所有表
    g.GenerateAllTable()

    g.Execute()
}

gormt

如果只想要结构体，不需要 Gen 那套查询代码，gormt 更轻量：

go install github.com/xxjwxc/gormt@latest
gormt -H 127.0.0.1 -d agenthub -u root -p 123456 -P 3306 -o ./internal/model

CRUD

Create

func Insert(db *gorm.DB, student Student) {
	result := db.Create(&student)
	if result.Error != nil {
		log.Fatalf("failed to create student: %v", result.Error)
	}
}

Read

func FindOne(db *gorm.DB) Student {
	var student Student
	result := db.First(&student, 1)		// search by primary key
	if result.Error != nil {
		log.Printf("failed to find student: %v", result.Error)
	}
	return student
}

func FindAll(db *gorm.DB) []Student {
	var students []Student
	result := db.Where("age > ?", 20).Find(&students)
	if result.Error != nil {
		log.Printf("failed to find students: %v", result.Error)
	}
	return students
}

事务

使用 Transaction 执行事务

if err := db.Transaction(func(tx *gorm.DB) error {
	if err := tx.Create(&student1).Error; err != nil {
		return err
	}
	if err := tx.Create(&student2).Error; err != nil {
		return err
	}
	return nil
}); err != nil {
       log.Fatalf("transaction failed: %v", err)

预处理和 SQL 注入

占位参数可保护数据值，但 Order、Table、Raw 等接收的 SQL 结构片段仍可能注入；动态列名和排序应使用白名单。链式调用本身不是安全保证。见 GORM 安全文档

var student Student
db.Where("name = ?", "Musuyin").First(&student)
fmt.Println(student)

示例

package main

import (
	"fmt"
	"log"

	"gorm.io/driver/mysql"
	"gorm.io/gorm"
)

// User 结构体映射到数据库中的users表
type User struct {
	ID   uint   `gorm:"primaryKey;column:id"` // ID为主键
	Name string `gorm:"size:255;column:name"` // Name字段最大长度为255
	Age  int    `gorm:"column:age"`           // Age字段
}

func main() {
	// 定义数据源名称
	dsn := "user:password@tcp(127.0.0.1:3306)/dbname?charset=utf8mb4&parseTime=True&loc=Local"

	// 连接数据库
	db, err := gorm.Open(mysql.Open(dsn), &gorm.Config{})
	if err != nil {
		log.Fatalf("failed to connect database: %v", err)
	}

	// 自动迁移User结构体
	db.AutoMigrate(&User{})

	// 创建一个新的User记录
	user := User{Name: "张三", Age: 25}
	result := db.Create(&user)
	if result.Error != nil {
		log.Fatalf("failed to create user: %v", result.Error)
	}

	// 查询ID为1的User记录
	var queriedUser User
	result = db.First(&queriedUser, user.ID) // 根据主键查询
	if result.Error != nil {
		log.Printf("failed to find user: %v", result.Error)
	} else {
		fmt.Printf("Queried User: %+v\n", queriedUser)
	}

	// 更新User记录的Age字段
	queriedUser.Age = 26
	result = db.Save(&queriedUser)
	if result.Error != nil {
		log.Fatalf("failed to update user: %v", result.Error)
	}

	// 删除User记录
	result = db.Delete(&queriedUser)
	if result.Error != nil {
		log.Fatalf("failed to delete user: %v", result.Error)
	}
}
