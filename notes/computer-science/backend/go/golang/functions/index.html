Go 函数

2025-02-15

#Go

函数把参数、返回值和控制流组合为可复用单元。闭包需要追踪捕获变量的生命周期，defer 需要分开看参数求值与函数执行时机，panic/recover 则需要沿同一 goroutine 的调用栈分析。

函数声明

// 基本形式
func name(parameters) returnType {
    // 函数体
}

// 示例
func add(a int, b int) int {
    return a + b
}

// 相同类型参数可合并
func add(a, b int) int {
    return a + b
}

// 无返回值
func printSum(a, b int) {
    fmt.Println(a + b)
}

// 多返回值
func divide(a, b float64) (float64, error) {
    if b == 0 {
        return 0, errors.New("division by zero")
    }
    return a / b, nil
}

// 命名返回值
func split(sum int) (x, y int) {
    x = sum * 4 / 9
    y = sum - x
    return  // 裸返回（返回命名变量）
}

可变参数

// 可变参数（必须是最后一个参数）
func sum(nums ...int) int {
    total := 0
    for _, n := range nums {
        total += n
    }
    return total
}

// 调用
sum(1, 2, 3)
sum(1, 2, 3, 4, 5)

// 传递切片
values := []int{1, 2, 3}
sum(values...)  // 展开切片

函数值

// 函数可以赋值给变量
func square(n int) int {
    return n * n
}

f := square
fmt.Println(f(3))  // 9

// 函数作为参数
func apply(f func(int) int, value int) int {
    return f(value)
}

result := apply(square, 5)  // 25

// 函数作为返回值
func multiplier(factor int) func(int) int {
    return func(x int) int {
        return x * factor
    }
}

double := multiplier(2)
triple := multiplier(3)
fmt.Println(double(5))  // 10
fmt.Println(triple(5))  // 15

匿名函数和闭包

// 匿名函数
func() {
    fmt.Println("Hello")
}()  // 立即执行

// 闭包：捕获外部变量
func counter() func() int {
    count := 0
    return func() int {
        count++
        return count
    }
}

c1 := counter()
fmt.Println(c1())  // 1
fmt.Println(c1())  // 2
fmt.Println(c1())  // 3

c2 := counter()
fmt.Println(c2())  // 1（独立的计数器）

闭包陷阱

// 错误：循环变量被捕获
funcs := []func(){}
for i := 0; i < 3; i++ {
    funcs = append(funcs, func() {
        fmt.Println(i)  // 捕获的是变量 i
    })
}
for _, f := range funcs {
    f()  // 输出: 3, 3, 3
}

// 正确方式1：创建副本
for i := 0; i < 3; i++ {
    i := i  // 创建新变量
    funcs = append(funcs, func() {
        fmt.Println(i)
    })
}

// 正确方式2：通过参数传递
for i := 0; i < 3; i++ {
    funcs = append(funcs, func(n int) func() {
        return func() { fmt.Println(n) }
    }(i))
}

defer 延迟执行

// defer 在函数返回前执行（后进先出）
func example() {
    defer fmt.Println("1")
    defer fmt.Println("2")
    defer fmt.Println("3")
    fmt.Println("body")
}
// 输出：body, 3, 2, 1

// 常用于资源清理
func readFile(filename string) error {
    f, err := os.Open(filename)
    if err != nil {
        return err
    }
    defer f.Close()  // 确保文件关闭

    // 读取文件...
    return nil
}

// defer 捕获参数值（不是引用）
func deferEval() {
    i := 1
    defer fmt.Println(i)  // 捕获 i=1
    i++
    // 输出: 1（不是 2）
}

// defer 修改返回值
func triple(x int) (result int) {
    defer func() {
        result *= 3
    }()
    return x + 1
}
fmt.Println(triple(2))  // 9（不是 3）

defer 使用场景

  1. 资源释放
mu.Lock()
defer mu.Unlock()
file.Open()
defer file.Close()
  1. 错误处理
func doSomething() (err error) {
    defer func() {
        if r := recover(); r != nil {
            err = fmt.Errorf("panic: %v", r)
        }
    }()
    // ...
}
  1. 性能追踪
func trace(name string) func() {
    start := time.Now()
    fmt.Printf("enter %s\n", name)
    return func() {
        fmt.Printf("exit %s (%s)\n", name, time.Since(start))
    }
}
func someFunc() {
    defer trace("someFunc")()
    // ...
}

defer 参数传递

  1. 传入数值
package main

import "fmt"

func deferRun() {
  var num = 1
  defer fmt.Printf("num is %d", num)

  num = 2
  return
}

func main(){
    deferRun()
}

defer 在被调用时即会将函数压入栈中，此时传入的参数就会固定了，所以即使后续修改了 num 的值，这里的输出仍为 1

  1. 传入指针
package main

import "fmt"

func main() {
 deferRun()
}

func deferRun() {
 var arr = [4]int{1, 2, 3, 4}
 defer printArr(&arr)

 arr[0] = 100
 return
}

func printArr(arr *[4]int) {
 for i := range arr {
  fmt.Println(arr[i])
 }
}

和普通函数传参一样，如果传入的是地址，那么在外部修改的结果也会影响内部，最后结果为 100 2 3 4

  1. 返回值
package main

import "fmt"

func main() {
   res := deferRun()
   fmt.Println(res)
}

func deferRun() (res int) {
  num := 1

  defer func() {
    res++
  }()

  return num
}

return 可以拆解为（即是一个非原子操作）：

  1. 设置返回值
  2. 执行 defer
  3. 返回结果

本例中，先另 res = num，再执行 defer 使 res++，最后返回结果为 2

  1. 返回值（匿名返回值）
func deferRun() (int) {
  num := 1

  defer func() {
    num++
  }()

  return num
}

本质上返回时，会创建一个临时变量存储返回值，区别在于此例返回值为匿名。当执行 return 时，可以假设此时创建一个返回值 ret = num，而之后即使另 num++，也不影响返回值 ret，所以最后结果仍为 1

panic 和 recover

panic

// panic 会立即停止当前函数，开始回溯调用栈
func mustOpen(filename string) *os.File {
    f, err := os.Open(filename)
    if err != nil {
        panic(err)  // 严重错误，程序无法继续
    }
    return f
}

// 内置的 panic 场景
// - 数组越界
// - 空指针解引用
// - 类型断言失败
// - 向已关闭的 channel 发送数据

recover

// recover 只在 defer 中有效
func safeDiv(a, b int) (result int, err error) {
    defer func() {
        if r := recover(); r != nil {
            err = fmt.Errorf("panic: %v", r)
        }
    }()
    return a / b, nil
}

result, err := safeDiv(10, 0)
if err != nil {
    fmt.Println(err)  // panic: runtime error: integer divide by zero
}

最佳实践：

  • 普通错误用 error 返回值
  • 严重的、不可恢复的错误用 panic
  • recover() 仅用于库的边界，避免崩溃传播
    • recover() 只能恢复当前函数级或以当前函数为首的调用链中函数的 panic()，恢复后当前函数结束，但上级函数继续执行
    • 函数发生 panic 之后会一直向上传递，直到被 recover() 捕获或者传递到 main() 使程序结束
