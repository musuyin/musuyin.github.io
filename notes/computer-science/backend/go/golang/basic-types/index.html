Go 基础数据类型

2025-02-15

#Go

基础类型决定数值范围、比较规则和编码方式。读数值示例时区分溢出与精度损失；读字符串示例时区分字节索引、Unicode 码点与用户看到的字符。

基础数据类型

Go 的基础数据类型分为四类：

  1. 基础类型：数字、字符串、布尔型
  2. 聚合类型：数组、结构体
  3. 引用类型：指针、切片、映射、函数、通道
  4. 接口类型

整型

整型类型

类型 说明 范围
int8 8 位有符号 -128 ~ 127
int16 16 位有符号 -32768 ~ 32767
int32 32 位有符号 -2^31 ~ 2^31-1
int64 64 位有符号 -2^63 ~ 2^63-1
uint8 8 位无符号 0 ~ 255
uint16 16 位无符号 0 ~ 65535

uintptr 指针大小的无符号整型 依平台而定

字面量表示

// 十进制
var dec int = 42

// 八进制（0 前缀）
var oct int = 052  // 42

// 十六进制（0x 或 0X 前缀）
var hex int = 0x2A  // 42
var hex2 int = 0X2a  // 42

// 二进制（0b 或 0B 前缀，Go 1.13+）
var bin int = 0b101010  // 42

// 下划线分隔（提高可读性，Go 1.13+）
var million int = 1_000_000
var binary int = 0b_1010_1010

运算符

算术运算符

+ - * / %        // 加 减 乘 除 取模
  • 除法 /：整数相除结果为整数（向零截断）
  • 取模 %：结果符号与被除数一致
fmt.Println(5 / 2)     // 2
fmt.Println(5.0 / 2)   // 2.5
fmt.Println(-5 % 3)    // -2（不是 1）
fmt.Println(5 % -3)    // 2（不是 -1）

比较运算符

== != < <= > >=

位运算符

&    // 按位与 AND
|    // 按位或 OR
^    // 按位异或 XOR（二元）或按位取反（一元）
&^   // 按位清除 AND NOT
<<   // 左移
>>   // 右移
var x uint8 = 0b1100_1010
var y uint8 = 0b0101_1100

fmt.Printf("%08b\n", x&y)   // 01000000 (与)
fmt.Printf("%08b\n", x|y)   // 11011110 (或)
fmt.Printf("%08b\n", x^y)   // 10011110 (异或)
fmt.Printf("%08b\n", ^x)    // 00110101 (取反)
fmt.Printf("%08b\n", x&^y)  // 10000010 (清除)

// 移位
fmt.Printf("%08b\n", x<<1)  // 10010100 (左移)
fmt.Printf("%08b\n", x>>1)  // 01100101 (右移)

位清除 &^ 说明：

z = x &^ y
// 等价于：对于 y 中为 1 的位，将 x 对应位置零
// 01100100 &^ 00111000 = 01000100

运算符优先级（从高到低）

类型转换

Go 要求显式类型转换：

var i int = 42
var f float64 = float64(i)
var u uint = uint(f)

// 不同整型间的转换
var i32 int32 = 100
var i64 int64 = int64(i32)

// 大类型转小类型可能溢出
var x int32 = 1<<31 - 1
var y int8 = int8(x)  // 溢出！y = -1

// 浮点转整型：截断小数部分，向零取整
var f1 float64 = 3.99
var i1 int = int(f1)  // 3
var f2 float64 = -3.99
var i2 int = int(f2)  // -3

常用操作

// 判断奇偶
if x%2 == 0 {
    fmt.Println("偶数")
}

// 位操作判断（更快）
if x&1 == 0 {
    fmt.Println("偶数")
}

// 2 的幂次判断
func isPowerOfTwo(x int) bool {
    return x > 0 && (x&(x-1)) == 0
}

// 统计二进制中 1 的个数
func countBits(x uint) int {
    count := 0
    for x != 0 {
        x &= x - 1  // 清除最右边的 1
        count++
    }
    return count
}

浮点数

类型

  • float32：IEEE 754 单精度（约 6 位十进制精度）
  • float64：IEEE 754 双精度（约 15 位十进制精度）

推荐使用 float64，因为精度更高，且 math 包的函数都使用 float64。

字面量

var f1 float64 = 3.14
var f2 float64 = .25      // 0.25
var f3 float64 = 2.       // 2.0
var f4 float64 = 1e9      // 1000000000
var f5 float64 = 1.5e-3   // 0.0015

特殊值

import "math"

var inf float64 = math.Inf(1)   // 正无穷 +Inf
var negInf = math.Inf(-1)        // 负无穷 -Inf
var nan = math.NaN()             // 非数 NaN

// 判断
fmt.Println(math.IsInf(inf, 1))  // true
fmt.Println(math.IsNaN(nan))     // true

// NaN 的特性
fmt.Println(nan == nan)          // false（NaN 不等于任何值，包括自己）
fmt.Println(nan < 1.0)           // false
fmt.Println(nan > 1.0)           // false

精度问题

// 浮点数不精确
var f float64 = 0.1 + 0.2
fmt.Println(f == 0.3)  // false
fmt.Println(f)         // 0.30000000000000004

// 比较浮点数应使用误差范围
func almostEqual(a, b, epsilon float64) bool {
    return math.Abs(a-b) < epsilon
}

fmt.Println(almostEqual(0.1+0.2, 0.3, 1e-9))  // true

格式化输出

var f float64 = 123.456789

fmt.Printf("%f\n", f)      // 123.456789（默认 6 位小数）
fmt.Printf("%.2f\n", f)    // 123.46（保留 2 位）
fmt.Printf("%e\n", f)      // 1.234568e+02（科学计数法）
fmt.Printf("%g\n", f)      // 123.457（紧凑格式）

复数

类型

  • complex64：实部和虚部都是 float32
  • complex128：实部和虚部都是 float64

创建和操作

// 字面量
var c1 complex128 = 1 + 2i
var c2 = 3 + 4i

// complex 函数
c3 := complex(1, 2)  // 1+2i

// 提取实部和虚部
real := real(c1)  // 1.0
imag := imag(c1)  // 2.0

// 运算
fmt.Println(c1 + c2)  // (4+6i)
fmt.Println(c1 * c2)  // (-5+10i)

// 比较
fmt.Println(c1 == complex(1, 2))  // true

布尔型

类型

bool：只有两个值 true 和 false

逻辑运算

&&  // 逻辑与（短路）
||  // 逻辑或（短路）
!   // 逻辑非
// 短路特性
func expensive() bool {
    fmt.Println("expensive called")
    return true
}

if false && expensive() {  // expensive 不会被调用
    // ...
}

if true || expensive() {   // expensive 不会被调用
    // ...
}

注意事项

// 布尔值不能隐式转换为数字
var b bool = true
// var i int = b  // 错误！

// 需要显式转换
func boolToInt(b bool) int {
    if b {
        return 1
    }
    return 0
}

// 数字也不能隐式转换为布尔
var n int = 1
// if n { }  // 错误！

// 必须显式比较
if n != 0 {
    // ...
}

字符串

基本概念

  • 字符串是不可变的字节序列
  • UTF-8 编码
  • 零值为 ""（空字符串）
  • 可以为空，但不会是 nil

字符串结构

// 内部结构（不可直接访问）
type stringStruct struct {
    str unsafe.Pointer  // 指向底层字节数组
    len int            // 长度（字节数）
}

字符串字面量

// 解释型字符串（双引号）
s1 := "Hello, 世界\n"  // 支持转义字符

// 原始字符串（反引号）
s2 := `Hello, 世界\n`  // 不转义，所见即所得
s3 := `多行
字符串
支持`

转义字符：

\n   // 换行
\r   // 回车
\t   // 制表符
\"   // 双引号
\\   // 反斜杠
\xhh // 十六进制字节
\uhhhh   // Unicode 码点（4 位十六进制）
\Uhhhhhhhh // Unicode 码点（8 位十六进制）

字符串操作

长度和索引

s := "Hello, 世界"

// 字节长度
fmt.Println(len(s))  // 13（不是 9！）

// 索引访问（字节）
fmt.Println(s[0])  // 72（'H' 的 ASCII 码）

// 子串（字节范围）
fmt.Println(s[0:5])   // "Hello"
fmt.Println(s[7:])    // "世界"（可能乱码，取决于位置）

// 字符数量（rune 个数）
fmt.Println(utf8.RuneCountInString(s))  // 9

拼接

// + 运算符
s1 := "Hello"
s2 := "World"
s3 := s1 + ", " + s2  // "Hello, World"

// strings.Join（推荐用于多个字符串）
parts := []string{"Hello", "World", "!"}
result := strings.Join(parts, " ")  // "Hello World !"

// fmt.Sprintf（格式化拼接）
name := "Alice"
age := 30
msg := fmt.Sprintf("%s is %d years old", name, age)

// strings.Builder（高效构建）
var builder strings.Builder
for i := 0; i < 100; i++ {
    builder.WriteString("Hello ")
}
result := builder.String()

性能对比：

  • + 适合少量拼接
  • strings.Builder 适合大量拼接（避免多次内存分配）

比较

s1 := "apple"
s2 := "banana"

// 字典序比较
fmt.Println(s1 == s2)  // false
fmt.Println(s1 < s2)   // true
fmt.Println(s1 > s2)   // false

// 忽略大小写比较
strings.EqualFold("Go", "go")  // true

遍历

s := "Hello, 世界"

// 按字节遍历
for i := 0; i < len(s); i++ {
    fmt.Printf("%d: %c\n", i, s[i])
}

// 按字符遍历（rune）
for i, r := range s {
    fmt.Printf("%d: %c\n", i, r)
}
// 输出：
// 0: H
// 1: e
// 2: l
// 3: l
// 4: o
// 5: ,
// 6:
// 7: 世  // 注意：索引是字节位置，不是字符位置
// 10: 界

UTF-8 编码

编码规则

  • ASCII 字符：1 字节（0xxxxxxx）
  • 其他字符：2-4 字节
    • 2 字节：110xxxxx 10xxxxxx
    • 3 字节：1110xxxx 10xxxxxx 10xxxxxx
    • 4 字节：11110xxx 10xxxxxx 10xxxxxx 10xxxxxx

rune 类型

// rune 是 int32 的别名，表示一个 Unicode 码点
var r rune = '世'
fmt.Printf("%c = U+%04X = %d\n", r, r, r)
// 世 = U+4E16 = 19990

// 字符字面量
c1 := 'A'       // rune (int32)
c2 := '\n'      // 换行符
c3 := '世'  // Unicode
c4 := '\U00004E16'

utf8 包

import "unicode/utf8"

s := "Hello, 世界"

// 字符数量
count := utf8.RuneCountInString(s)  // 9

// 解码第一个字符
r, size := utf8.DecodeRuneInString(s)
// r = 'H', size = 1

// 判断是否为有效 UTF-8
valid := utf8.ValidString(s)  // true

// 编码 rune
buf := make([]byte, 4)
n := utf8.EncodeRune(buf, '世')  // n = 3

字符串和字节切片

// 字符串 -> []byte
s := "Hello"
b := []byte(s)  // 复制数据

// []byte -> 字符串
s2 := string(b)  // 复制数据

// 性能提示：频繁转换开销大，尽量减少

strings 包常用函数

import "strings"

s := "Hello, World! Hello!"

// 查找
strings.Contains(s, "World")       // true
strings.Index(s, "World")          // 7
strings.LastIndex(s, "Hello")      // 14
strings.Count(s, "l")              // 5

// 前缀/后缀
strings.HasPrefix(s, "Hello")      // true
strings.HasSuffix(s, "!")          // true

// 分割
parts := strings.Split("a,b,c", ",")  // ["a", "b", "c"]
fields := strings.Fields("a  b  c")    // ["a", "b", "c"]

// 替换
strings.Replace(s, "Hello", "Hi", 1)   // "Hi, World! Hello!"
strings.ReplaceAll(s, "Hello", "Hi")   // "Hi, World! Hi!"

// 大小写
strings.ToUpper(s)   // "HELLO, WORLD! HELLO!"
strings.ToLower(s)   // "hello, world! hello!"

// 修剪
strings.Trim("  hello  ", " ")     // "hello"
strings.TrimPrefix("hello", "hel")  // "lo"

strconv 包

import "strconv"

// 整数 <-> 字符串
i, err := strconv.Atoi("42")        // 42
s := strconv.Itoa(42)               // "42"

// 指定进制
i, err = strconv.ParseInt("1010", 2, 64)  // 10 (二进制)
s = strconv.FormatInt(42, 2)              // "101010"

// 浮点数 <-> 字符串
f, err := strconv.ParseFloat("3.14", 64)  // 3.14
s = strconv.FormatFloat(3.14, 'f', 2, 64) // "3.14"

// 布尔 <-> 字符串
b, err := strconv.ParseBool("true")   // true
s = strconv.FormatBool(true)          // "true"

常量声明

// 单个常量
const Pi = 3.14159
const MaxSize = 1000

// 批量声明
const (
    StatusOK    = 200

const TypedPi float64 = 3.14159

iota 常量生成器

iota 是常量计数器，从 0 开始，每行递增 1。

// 基本用法
const (
    Sunday = iota     // 0
    Monday            // 1
    Tuesday           // 2
    Wednesday         // 3
    Thursday          // 4
    Friday            // 5
    Saturday          // 6
)

// 跳过值
const (
    _ = iota          // 跳过 0
    KB = 1 << (10 * iota)  // 1024
    MB                     // 1048576
    GB                     // 1073741824
    TB
)

// 复杂表达式
const (
    FlagNone = 1 << iota  // 1
    FlagRed               // 2
    FlagGreen             // 4
    FlagBlue              // 8
)

// 多个 iota
const (
    A, B = iota, iota * 10  // 0, 0
    C, D                     // 1, 10
    E, F                     // 2, 20
)

无类型常量

Go 的常量可以是无类型的，这提供了更高的精度和灵活性。

// 无类型常量
const Pi = 3.14159265358979323846264338327950288419716939937510

// 可以直接用于不同类型
var f32 float32 = Pi  // 自动转换为 float32
var f64 float64 = Pi  // 自动转换为 float64
var c128 complex128 = Pi  // 自动转换为 complex128

// 无类型整数常量可以表示任意大的数
const Huge = 1 << 100  // 合法！
// var x int = Huge    // 错误！溢出

// 常量算术（编译时计算）
const (
    GiB = 1 << 30
    TiB = 1 << 40
    result = TiB / GiB  // 1024
)

常量的类型推断

// 常量字面量的默认类型
const (
    i = 0       // untyped int
    f = 0.0     // untyped float
    c = 0i      // untyped complex
    b = true    // untyped bool
    s = "text"  // untyped string
    r = 'A'     // untyped rune
)

// 赋值时的隐式转换
var f64 float64 = 3  // int -> float64
var c128 complex128 = 3.14  // float -> complex128
