Go 接口

2025-02-15

#Go

接口用行为约束解耦调用者与具体实现。判断一个接口值时，要同时看动态类型与动态值；这也是理解 typed nil、类型断言与比较行为的共同入口。

接口定义

接口是方法的集合，定义了对象的行为。

// 定义接口
type Reader interface {
    Read(p []byte) (n int, err error)
}

type Writer interface {
    Write(p []byte) (n int, err error)
}

// 接口组合
type ReadWriter interface {
    Reader
    Writer
}

// 等价于
type ReadWriter interface {
    Read(p []byte) (n int, err error)
    Write(p []byte) (n int, err error)
}

接口实现

Go 的接口实现是隐式的（Duck Typing）。

type File struct {
    // ...
}

// File 实现了 Writer 接口（不需要显式声明）
func (f *File) Write(p []byte) (n int, err error) {
    // 实现...
    return len(p), nil
}

// 使用
var w Writer = new(File)  // File 实现了 Writer
w.Write([]byte("hello"))

面试追问：空接口与非空接口

空接口（interface{} / any）的方法集为空，任何类型都能赋值；运行时主要保存动态类型和数据指针。非空接口要求动态类型实现其方法集，运行时还携带与该接口方法表相关的信息，调用可动态派发。二者都可能出现“动态值为 nil 指针但接口本身非 nil”的陷阱。Go 1.18 后优先用泛型表达类型参数约束，避免把 any 当作无类型容器。

空接口

interface{} 可以持有任意类型的值。

var any interface{}

any = 42
any = "hello"
any = struct{ Name string }{"Alice"}

// 常用于泛型容器
func PrintAny(v interface{}) {
    fmt.Println(v)
}

// 类型断言
if str, ok := any.(string); ok {
    fmt.Println("字符串:", str)
}

类型断言

var w io.Writer = os.Stdout

// 类型断言：判断动态类型
f, ok := w.(*os.File)
if ok {
    // w 的动态类型是 *os.File
    fmt.Println(f)
}

// 不安全的断言（失败会 panic）
f := w.(*os.File)

// 类型 switch
func classify(v interface{}) {
    switch x := v.(type) {
    case nil:
        fmt.Println("nil")
    case int:
        fmt.Printf("整数: %d\n", x)
    case string:
        fmt.Printf("字符串: %s\n", x)
    case bool:
        fmt.Printf("布尔: %t\n", x)
    default:
        fmt.Printf("未知类型: %T\n", x)
    }
}

接口值

接口值由类型和值两部分组成。

var w io.Writer

// nil 接口
fmt.Printf("%T, %v\n", w, w)  // <nil>, <nil>

// 具体类型的值
w = os.Stdout
fmt.Printf("%T, %v\n", w, w)  // *os.File, &{...}

// nil 指针但接口非 nil
var buf *bytes.Buffer
w = buf
fmt.Printf("%T, %v\n", w, w)  // *bytes.Buffer, <nil>
fmt.Println(w == nil)         // false（接口不是 nil）

// 正确的 nil 检查
if w != nil {
    if v := reflect.ValueOf(w); v.Kind() == reflect.Ptr && v.IsNil() {
        fmt.Println("值是 nil 指针")
    }
}

接口比较

var w1, w2 io.Writer

w1 = os.Stdout
w2 = os.Stdout
fmt.Println(w1 == w2)  // true

w1 = new(bytes.Buffer)
w2 = new(bytes.Buffer)
fmt.Println(w1 == w2)  // false（不同的 Buffer 实例）

// 注意：如果动态类型不可比较，会 panic
var x interface{} = []int{1, 2, 3}
// fmt.Println(x == x)  // panic!

接口赋值

var phone Phone      // 声明一个接口类型phone
phone = new(Apple)   // 注意这种创建方式，new函数参数是接口的实现
phone.(*Apple).PhoneName = "Apple"  // 使用断言给phone的成员赋值

注意需要给接口赋值指针，才能修改内部的属性（左值）；如果是赋值普通结构体，则通过断言获取的是右值，无法修改

常用接口

fmt.Stringer

type Stringer interface {
    String() string
}

type Point struct{ X, Y int }

func (p Point) String() string {
    return fmt.Sprintf("(%d, %d)", p.X, p.Y)
}

p := Point{1, 2}
fmt.Println(p)  // 自动调用 String() 方法

error 接口

// errors
type error interface {
    Error() string
}

func New(text string) error {
    return &errorString{text}
}

// fmt
func Errorf(format string, a ...interface{}) error {
   p := newPrinter()
   p.wrapErrs = true
   p.doPrintf(format, a)
   s := string(p.buf)
   var err error
   if p.wrappedErr == nil {
      err = errors.New(s)
   } else {
      err = &wrapError{s, p.wrappedErr}
   }
   p.free()
   return err
}

// self package
// 自定义错误类型
type MyError struct {
    Code    int
    Message string
}

// Error 实现 error 接口
func (e *MyError) Error() string {
    return fmt.Sprintf("错误 %d: %s", e.Code, e.Message)
}

func doSomething() error {
    return &MyError{Code: 404, Message: "not found"}
}

errors.New() 每次创建独立错误值；error 接口可以比较，但底层动态类型必须可比较。识别包装错误或哨兵错误优先用 errors.Is，见 errors 文档。

sort.Interface

type Interface interface {
    Len() int
    Less(i, j int) bool
    Swap(i, j int)
}

// 实现自定义排序
type ByLength []string

func (s ByLength) Len() int           { return len(s) }
func (s ByLength) Less(i, j int) bool { return len(s[i]) < len(s[j]) }
func (s ByLength) Swap(i, j int)      { s[i], s[j] = s[j], s[i] }

// 使用
words := []string{"Go", "Python", "C", "JavaScript"}
sort.Sort(ByLength(words))
fmt.Println(words)  // [C Go Python JavaScript]

接口设计原则

  1. 接口隔离原则：接口应该小而专注
   // 好：单一职责
   type Reader interface { Read(p []byte) (n int, err error) }
   type Writer interface { Write(p []byte) (n int, err error) }

   // 不好：职责过多
   type ReadWriteCloseFlusher interface {
       Read(p []byte) (n int, err error)
       Write(p []byte) (n int, err error)
       Close() error
       Flush() error
   }
  1. 接受接口，返回具体类型
   // 好：参数是接口，返回具体类型
   func ReadAll(r io.Reader) ([]byte, error)

   // 不好：返回接口
   func NewReader() io.Reader  // 调用者难以获取具体类型
  1. 避免过早抽象
    • 先写具体实现，再根据需要提取接口
    • “接口属于使用者，而非实现者”
